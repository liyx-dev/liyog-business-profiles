/* =====================================================================
   LIYOG WORLD BRAND PROFILES — boost-ui.js   (3a fix release)
   Boost purchase sheet. Scope (profile / catalogue / product) is decided
   by the CALLER (profile.js). Exposes exactly one entry point,
   window.LiyogBoostUI.openBoostSheet(p, options) — unchanged signature.

   WHAT CHANGED IN THIS RELEASE:
   - The browser now sends only INTENT to the server:
       country_mode ("own"|"all"), category_mode ("own"|"all"), show_cta.
     The real country / category values are derived on the server from
     the profile row. No targeting values live in this file.
   - Own country / category come from the pricing response (same single
     request as before — no extra network call).
   - No country on the profile -> "All countries" + a clear note telling
     the user to update their profile country. Same safe handling for
     category and for contact buttons (needs a WhatsApp/phone number).
   - New premium UI: segmented controls with an obvious active state,
     a real on/off switch for the contact buttons (with a live preview),
     and a live "Your boost will show to" summary.
   - New review/recap step before payment, an authoritative recap for
     manual payments, and the targeting is now INCLUDED in the WhatsApp
     message.
   - After returning from Paystack: a visible result toast + a recap
     sheet (needs GET /api/boost/recap — see boost.js notes; degrades
     gracefully to just the toast if that route isn't added).
   - Toasts are now actually visible (the old .lp-toast-host had no CSS
     and the toast never received .lp-toast-visible, so it was invisible).
   - Body scroll lock now uses profile.js's shared reference-counted lock
     when available (fixes scroll freezing under stacked sheets).
   ===================================================================== */

  function esc(str) {
    return String(str == null ? "" : str).replace(/[&<>"']/g, (c) => ({
      "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
    }[c]));
  }

  // Prefer profile.js's shared reference-counted lock (exposed on
  // window.LiyogProductUI) so stacked sheets never leave scroll frozen.
  function lockBodyScroll() {
    const ui = window.LiyogProductUI;
    if (ui && ui.lockBodyScroll) { ui.lockBodyScroll(); return; }
    document.body.style.overflow = "hidden";
  }
  function unlockBodyScroll() {
    const ui = window.LiyogProductUI;
    if (ui && ui.unlockBodyScroll) { ui.unlockBodyScroll(); return; }
    const somethingStillOpen =
      document.querySelector(".lp-sheet-overlay.lp-open") ||
      document.querySelector(".lp-pm-overlay.lp-open") ||
      document.querySelector(".lp-product-lightbox.lp-open");
    if (!somethingStillOpen) document.body.style.overflow = "";
  }
  function openSheet(overlay) {
    if (overlay.classList.contains("lp-open")) return; // never double-lock
    overlay.classList.add("lp-open");
    lockBodyScroll();
  }
  function closeSheet(overlay) {
    if (!overlay.classList.contains("lp-open")) return;
    overlay.classList.remove("lp-open");
    unlockBodyScroll();
  }

  function showToast(message, type) {
    type = type || "success";
    const ui = window.LiyogProductUI;
    if (ui && ui.showToast) { ui.showToast(message, type); return; }

    let host = document.getElementById("lpb-toast-host");
    if (!host) {
      host = document.createElement("div");
      host.id = "lpb-toast-host";
      document.body.appendChild(host);
    }
    const toast = document.createElement("div");
    toast.className = "lpb-toast" + (type === "error" ? " error" : "");
    toast.textContent = message;
    host.appendChild(toast);
    requestAnimationFrame(() => toast.classList.add("show"));
    setTimeout(() => {
      toast.classList.remove("show");
      setTimeout(() => toast.remove(), 300);
    }, 4200);
  }

  function formatPrice(amountMinorUnits, currency) {
    const major = amountMinorUnits / 100;
    const symbols = { NGN: "₦", GHS: "GH₵", ZAR: "R", KES: "KSh", USD: "$" };
    const symbol = symbols[currency] || currency + " ";
    return `${symbol}${major.toLocaleString(undefined, { maximumFractionDigits: 0 })}`;
  }

  function formatDuration(hours) {
    const h = Number(hours) || 0;
    if (h < 24) return `${h} hour${h === 1 ? "" : "s"}`;
    const d = h / 24;
    return Number.isInteger(d) ? `${d} day${d === 1 ? "" : "s"}` : `${h} hours`;
  }

  function prettify(slug) {
    return String(slug || "").replace(/[_-]+/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
  }

  async function fetchBoostPricing(profileId) {
    const qs = profileId ? `?profile_id=${encodeURIComponent(profileId)}` : "";
    const res = await fetch(`${WORKER_BASE}/api/boost/pricing${qs}`);
    if (!res.ok) throw new Error("Pricing fetch failed");
    return res.json();
  }

  function freshState() { return { countryMode: "own", categoryMode: "own", showCta: false }; }

  // Used only if the server didn't return a `targeting` block (e.g. the
  // worker isn't updated yet). The server stays the real authority.
  function fallbackInfo(p) {
    const channels = [];
    if (p.whatsapp_number) channels.push("whatsapp");
    if (p.phone_number) channels.push("call");
    return {
      ownCountry: null,
      countryNote: "We couldn't find a country on your profile, so this boost will show to all countries. To show it only in your own country, update the Country field on your profile.",
      ownCategory: p.business_category ? { slug: p.business_category, label: prettify(p.business_category) } : null,
      categoryNote: "We couldn't find a category on your profile, so this boost will show in all categories.",
      ctaAvailable: channels.length > 0,
      ctaChannels: channels,
      ctaNote: "Add a WhatsApp or phone number to your profile to show contact buttons on your sponsored cards."
    };
  }

  function getInfo(overlay, p) { return overlay.__lpBoostInfo || fallbackInfo(p); }

  function effectiveSelection(info, state) {
    const countryOwn = state.countryMode === "own" && info.ownCountry;
    const categoryOwn = state.categoryMode === "own" && info.ownCategory;
    const cta = !!(state.showCta && info.ctaAvailable);
    return {
      countryLabel: countryOwn ? info.ownCountry.name : "All countries",
      categoryLabel: categoryOwn ? info.ownCategory.label : "All categories",
      cta,
      ctaLabel: cta ? "Shown on your cards (" + ctaChannelText(info.ctaChannels) + ")" : "Not shown"
    };
  }

  function ctaChannelText(channels) {
    const names = (channels || []).map((c) => (c === "whatsapp" ? "WApp" : "Call"));
    return names.length ? names.join(" + ") : "WApp / Call";
  }

  // Only INTENT is sent — the server derives every real value.
  function payloadFromState(info, state) {
    return {
      country_mode: info.ownCountry ? state.countryMode : "all",
      category_mode: info.ownCategory ? state.categoryMode : "all",
      show_cta: !!(state.showCta && info.ctaAvailable)
    };
  }

   * p = profile object (needs p.id, p.business_name, p.slug).
   * options = { scope, product, keepState }.
   */
  function openBoostSheet(p, options) {
    const scope = (options && options.scope) || "profile";
    const product = (options && options.product) || null;
    const keepState = !!(options && options.keepState);

    const host = getOverlayHost();
    let overlay = document.getElementById("lp-boost-overlay");
    if (!overlay) {
      overlay = document.createElement("div");
      overlay.className = "lp-sheet-overlay";
      overlay.id = "lp-boost-overlay";
      host.appendChild(overlay);
      overlay.addEventListener("click", (e) => { if (e.target === overlay) closeSheet(overlay); });
    }

    if (!keepState || !overlay.__lpBoostState) overlay.__lpBoostState = freshState();
    if (!keepState) overlay.__lpBoostInfo = null;

    const scopeLabel = scope === "product" && product ? `"${esc(product.name)}"` : SCOPE_LABELS[scope];

    overlay.innerHTML = `
      <div class="lp-sheet">
        <div class="lp-sheet-top-bar">
          <div class="lp-sheet-handle"></div>
          <button class="lp-sheet-close" id="lp-boost-close" type="button" aria-label="Close">
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg>
          </button>
        </div>
        <div class="lp-boost-sheet-content">
          <h3 class="lpb-title">Boost ${scopeLabel}</h3>
          <p class="lpb-sub">Get featured placement and stand out to visitors. Choose your audience, then how long to boost.</p>
          <div id="lp-boost-targeting"><div class="lpb-skeleton"></div></div>
          <div id="lp-boost-durations" class="lp-tiers-list">
            <p style="text-align:center;color:var(--ink-soft, #666);font-size:13px;grid-column:1/-1;">Loading pricing…</p>
          </div>
        </div>
      </div>
    `;
    document.getElementById("lp-boost-close").addEventListener("click", () => closeSheet(overlay));
    openSheet(overlay);

    // Store scope/product so later steps can read them back.
    overlay.dataset.scope = scope;
    if (product) overlay.dataset.productId = product.id;

    fetchBoostPricing(p.id)
      .then((data) => {
        overlay.__lpBoostInfo = data.targeting || null;
        renderTargeting(overlay, p);

        const listEl = document.getElementById("lp-boost-durations");
        if (!listEl) return;

        if (!data.durations || !data.durations.length) {
          listEl.innerHTML = `<p style="text-align:center;color:var(--ink-soft, #666);font-size:13px;grid-column:1/-1;">Boosting isn't available right now — please check back soon.</p>`;
          return;
        }

        listEl.innerHTML = data.durations.map(durationCardHtml).join("") + customDurationHtml(data.customDuration);

        listEl.querySelectorAll("[data-duration-id]").forEach((card) => {
          card.addEventListener("click", () => {
            openMethodChoice(overlay, p, scope, product, {
              durationId: card.dataset.durationId,
              amount: Number(card.dataset.amount),
              currency: card.dataset.currency,
              label: DURATION_LABELS[card.dataset.durationId] || card.dataset.durationId
            });
          });
        });

        wireCustomDurationInput(listEl, data.durations, data.customDuration);
      })
      .catch(() => {
        renderTargeting(overlay, p);
        const listEl = document.getElementById("lp-boost-durations");
        if (listEl) listEl.innerHTML = `<p style="text-align:center;color:var(--ink-soft, #666);font-size:13px;grid-column:1/-1;">Couldn't load pricing — check your connection and try again.</p>`;
      });
  }

  function durationCardHtml(d) {
    const label = DURATION_LABELS[d.duration_id] || d.duration_id;
    return `
      <button type="button" class="lp-tier-card" data-duration-id="${esc(d.duration_id)}" data-amount="${esc(d.amount)}" data-currency="${esc(d.currency)}">
        <div class="lp-tier-card-name">${esc(label)}</div>
        <div class="lp-tier-card-price">${esc(formatPrice(d.amount, d.currency))}</div>
      </button>
    `;
  }

  function customDurationHtml(customDuration) {
    if (!customDuration) return "";
    const { minDays, maxDays } = customDuration;
    return `
      <div class="lp-boost-custom-duration">
        <label for="lp-boost-custom-days" class="lp-boost-custom-label">Need longer? Enter a custom duration (${Math.ceil(minDays)}–${maxDays} days)</label>
        <div class="lp-boost-custom-row">
          <input type="number" id="lp-boost-custom-days" min="${Math.ceil(minDays)}" max="${maxDays}" placeholder="e.g. 60" class="lp-field-input lp-boost-custom-input">
          <span class="lp-boost-custom-unit">days</span>
        </div>
        <p class="lp-boost-custom-hint">For serious sellers — bulk pricing kicks in the longer you go.</p>
        <p id="lp-boost-custom-preview" class="lp-boost-custom-preview"></p>
        <button type="button" class="lp-btn lp-btn-outline-subtle" id="lp-boost-custom-continue" style="width:100%;margin-top:10px;" disabled>
          <span>Continue with Custom Duration</span>
        </button>
      </div>
    `;
  }

  function wireCustomDurationInput(listEl, durations, customDuration) {
    const input = document.getElementById("lp-boost-custom-days");
    const preview = document.getElementById("lp-boost-custom-preview");
    const continueBtn = document.getElementById("lp-boost-custom-continue");
    if (!input || !customDuration) return;

    let debounceTimer = null;
    const { minDays, maxDays } = customDuration;

    input.addEventListener("input", () => {
      clearTimeout(debounceTimer);
      const days = Number(input.value);
      if (!days || days < minDays) {
        preview.textContent = days ? `Minimum is ${Math.ceil(minDays)} days — pick a listed duration for anything shorter.` : "";
        preview.className = days ? "lp-boost-custom-preview lp-boost-custom-preview-error" : "lp-boost-custom-preview";
        continueBtn.disabled = true;
        return;
      }
      if (days > maxDays) {
        preview.textContent = `Maximum is ${maxDays} days.`;
        preview.className = "lp-boost-custom-preview lp-boost-custom-preview-error";
        continueBtn.disabled = true;
        return;
      }
      debounceTimer = setTimeout(() => {
        preview.textContent = "Calculating…";
        preview.className = "lp-boost-custom-preview";
        try {
          // Preview only — the server recalculates the real price.
          const amount = estimateCustomPrice(durations, days, minDays - 1, maxDays);
          const currency = durations[0].currency;
          preview.textContent = `Estimated: ${formatPrice(amount, currency)}`;
          continueBtn.disabled = false;
          continueBtn.dataset.days = days;
          continueBtn.dataset.amount = amount;
          continueBtn.dataset.currency = currency;
        } catch (e) {
          preview.textContent = "Couldn't calculate — try a listed duration instead.";
          preview.className = "lp-boost-custom-preview lp-boost-custom-preview-error";
          continueBtn.disabled = true;
        }
      }, 350);
    });

    continueBtn.addEventListener("click", () => {
      continueBtn.dispatchEvent(new CustomEvent("lp-boost-custom-chosen", {
        bubbles: true,
        detail: {
          customDays: Number(continueBtn.dataset.days),
          amount: Number(continueBtn.dataset.amount),
          currency: continueBtn.dataset.currency
        }
      }));
    });
  }

  function estimateCustomPrice(durations, customDays, longestDays, maxDays, maxDiscountPct = 35) {
    const sorted = [...durations].sort((a, b) => a.hours - b.hours);
    const longest = sorted[sorted.length - 1];
    const longestTierDays = longest.hours / 24;
    const baseRate = longest.amount / longestTierDays;
    const denominator = Math.max(1, maxDays - longestTierDays);
    const progress = Math.min(1, (customDays - longestTierDays) / denominator);
    const discount = (maxDiscountPct / 100) * progress;
    const effectiveRate = baseRate * (1 - discount);
    return Math.ceil(effectiveRate * customDays);
  }

  // ------------------------------------------------------------------
  // Recap rendering (used before payment, for manual, and after Paystack)
  // ------------------------------------------------------------------
  function recapRowsHtml(rows) {
    return `<div class="lpb-recap">${rows.map((r) =>
      `<div class="lpb-recap-row ${r.total ? "lpb-recap-total" : ""}"><span class="lpb-recap-key">${esc(r.key)}</span><span class="lpb-recap-val">${esc(r.value)}</span></div>`
    ).join("")}</div>`;
  }

  /** Server recap object -> display rows. */
  function rowsFromServerRecap(recap, scopeText, durationText, priceText, extra) {
    const countries = recap.countries === "ALL" ? "All countries" : (recap.countries || []).map((c) => c.name).join(", ");
    const categories = recap.categories === "ALL" ? "All categories" : (recap.categories || []).map((c) => c.label).join(", ");
    const cta = recap.showCta ? "Shown on your cards (" + ctaChannelText(recap.ctaChannels) + ")" : "Not shown";
    const rows = [
      { key: "Boosting", value: scopeText },
      { key: "Duration", value: durationText },
      { key: "Country", value: countries },
      { key: "Category", value: categories },
      { key: "Contact buttons", value: cta }
    ];
    if (extra && extra.length) rows.push(...extra);
    rows.push({ key: "Price", value: priceText, total: true });
    return rows;
  }

  function scopeTextFor(scope, productName) {
    if (scope === "product" && productName) return `Product: ${productName}`;
    return SCOPE_LABELS[scope] || scope;
  }

  function notesHtml(notes) {
    return (notes || []).filter(Boolean).map((n) => `<div class="lpb-note">${ICON.info}<span>${esc(n)}</span></div>`).join("");
  }

  // ------------------------------------------------------------------
  // Review + payment step
  // ------------------------------------------------------------------
  function openMethodChoice(overlay, p, scope, product, duration) {
    const info = getInfo(overlay, p);
    const state = overlay.__lpBoostState || freshState();
    const sel = effectiveSelection(info, state);
    const payload = payloadFromState(info, state);
    const priceText = formatPrice(duration.amount, duration.currency);

    const rows = [
      { key: "Boosting", value: scopeTextFor(scope, product && product.name) },
      { key: "Duration", value: duration.label },
      { key: "Country", value: sel.countryLabel },
      { key: "Category", value: sel.categoryLabel },
      { key: "Contact buttons", value: sel.ctaLabel },
      { key: "Price", value: priceText, total: true }
    ];

    const pendingNotes = [];
    if (!info.ownCountry) pendingNotes.push(info.countryNote);
    if (!info.ownCategory) pendingNotes.push(info.categoryNote);

    overlay.querySelector(".lp-boost-sheet-content").innerHTML = `
      <button type="button" id="lp-boost-back" class="lpb-back">&larr; Back</button>
      <h3 class="lpb-title">Review your boost</h3>
      <p class="lpb-sub">Check the details, then choose how to pay.</p>
      ${recapRowsHtml(rows)}
      ${notesHtml(pendingNotes)}
      <div style="height:14px"></div>
      <button type="button" class="lp-btn lp-btn-primary lpb-btn-full" id="lp-boost-pay-paystack" style="margin-bottom:10px;">
        <span>Pay Now — Card / Bank</span>
      </button>
      <button type="button" class="lp-btn lp-btn-outline-subtle lpb-btn-full" id="lp-boost-pay-manual">
        <span>Pay via WhatsApp (Manual)</span>
      </button>
    `;

    document.getElementById("lp-boost-back").addEventListener("click", () => openBoostSheet(p, { scope, product, keepState: true }));

    const baseBody = () => ({
      profile_id: p.id,
      scope,
      product_id: product ? product.id : null,
      duration_id: duration.durationId || null,
      custom_days: duration.customDays || null,
      ...payload
    });

    document.getElementById("lp-boost-pay-paystack").addEventListener("click", async (e) => {
      const btn = e.currentTarget;
      btn.disabled = true;
      btn.querySelector("span").textContent = "Starting checkout…";
      try {
        const res = await fetch(`${WORKER_BASE}/api/boost/checkout`, {
          method: "POST",
          credentials: "include",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ ...baseBody(), method: "paystack" })
        });
        const data = await res.json();
        if (!res.ok || !data.success) {
          showToast(data.error || "Couldn't start payment. Please try again.", "error");
          btn.disabled = false;
          btn.querySelector("span").textContent = "Pay Now — Card / Bank";
          return;
        }
        window.location.href = data.authorizationUrl;
      } catch (err) {
        showToast("Couldn't start payment. Please check your connection.", "error");
        btn.disabled = false;
        btn.querySelector("span").textContent = "Pay Now — Card / Bank";
      }
    });

    document.getElementById("lp-boost-pay-manual").addEventListener("click", async (e) => {
      const btn = e.currentTarget;
      btn.disabled = true;
      try {
        const res = await fetch(`${WORKER_BASE}/api/boost/checkout`, {
          method: "POST",
          credentials: "include",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ ...baseBody(), method: "manual" })
        });
        const data = await res.json();
        if (!res.ok || !data.success) {
          showToast(data.error || "Couldn't start this — please try again.", "error");
          btn.disabled = false;
          return;
        }
        openManualHandoff(overlay, p, scope, product, data.purchase, duration);
      } catch (err) {
        showToast("Couldn't start this — please check your connection.", "error");
        btn.disabled = false;
      }
    });
  }

  // ------------------------------------------------------------------
  // Manual handoff — shows the SERVER's recap (what was actually stored)
  // and puts the same details in the WhatsApp message.
  // ------------------------------------------------------------------
  function openManualHandoff(overlay, p, scope, product, purchase, duration) {
    const priceText = formatPrice(purchase.amount, purchase.currency);
    const recap = purchase.recap || null;
    const scopeText = scopeTextFor(scope, product && product.name);
    const durationText = formatDuration(purchase.hours);

    let rows;
    let notes = [];
    if (recap) {
      rows = rowsFromServerRecap(recap, scopeText, durationText, priceText, [{ key: "Reference", value: purchase.id.slice(0, 8).toUpperCase() }]);
      notes = recap.notes || [];
    } else {
      rows = [
        { key: "Boosting", value: scopeText },
        { key: "Duration", value: durationText },
        { key: "Reference", value: purchase.id.slice(0, 8).toUpperCase() },
        { key: "Price", value: priceText, total: true }
      ];
    }

    overlay.querySelector(".lp-boost-sheet-content").innerHTML = `
      <h3 class="lpb-title">Almost done</h3>
      <p class="lpb-sub">Here's what we've saved for your boost. Message us on WhatsApp with your payment and we'll activate it shortly after.</p>
      ${recapRowsHtml(rows)}
      ${notesHtml(notes)}
      <div style="height:14px"></div>
      <button type="button" class="lp-btn lp-btn-primary lpb-btn-full" id="lp-boost-manual-whatsapp">
        <span>Message Us on WhatsApp</span>
      </button>
    `;

    document.getElementById("lp-boost-manual-whatsapp").addEventListener("click", async () => {
      const countries = recap ? (recap.countries === "ALL" ? "All countries" : (recap.countries || []).map((c) => c.name).join(", ")) : "—";
      const categories = recap ? (recap.categories === "ALL" ? "All categories" : (recap.categories || []).map((c) => c.label).join(", ")) : "—";
      const cta = recap ? (recap.showCta ? "Yes (" + ctaChannelText(recap.ctaChannels) + ")" : "No") : "—";

      const lines = [
        `I'd like to boost ${scope === "product" && product ? `"${product.name}"` : SCOPE_LABELS[scope]}.`,
        `Brand: ${p.business_name} (${WORKER_BASE}/b/${p.slug})`,
        `Duration: ${durationText}`,
        `Amount: ${priceText}`,
        `Country targeting: ${countries}`,
        `Category targeting: ${categories}`,
        `Contact buttons on cards: ${cta}`,
        `Purchase ID: ${purchase.id}`
      ];
      const waText = lines.join("\n");

      let adminNumber = "";
      try {
        const cfgRes = await fetch(`${WORKER_BASE}/api/boost-config`);
        const cfg = await cfgRes.json();
        adminNumber = cfg.adminWhatsapp || "";
      } catch (e) { /* wa.me still opens a compose screen without a number */ }

      window.open(`https://wa.me/${adminNumber}?text=${encodeURIComponent(waText)}`, "_blank");
      closeSheet(overlay);
    });
  }

  // Custom-duration "Continue" event -> same review step as listed durations.
  document.addEventListener("lp-boost-custom-chosen", (e) => {
    const overlay = document.getElementById("lp-boost-overlay");
    if (!overlay || !overlay.__lpBoostContext) return;
    const { p, scope, product } = overlay.__lpBoostContext;
    openMethodChoice(overlay, p, scope, product, {
      customDays: e.detail.customDays,
      amount: e.detail.amount,
      currency: e.detail.currency,
      label: `${e.detail.customDays} Days (Custom)`
    });
  });

  // Public entry point wrapper (unchanged signature): resets the
  // targeting state for every fresh open from outside, and stashes the
  // context the custom-duration listener above needs.
  function openBoostSheetWithContext(p, options) {
    openBoostSheet(p, { scope: options && options.scope, product: options && options.product, keepState: false });
    const overlay = document.getElementById("lp-boost-overlay");
    if (overlay) {
      overlay.__lpBoostContext = {
        p,
        scope: (options && options.scope) || "profile",
        product: (options && options.product) || null
      };
    }
  }

  // ------------------------------------------------------------------
  // Returning from Paystack: visible toast + recap of what was paid for.
  // ------------------------------------------------------------------
  function waitForHost(cb) {
    let tries = 0;
    const timer = setInterval(() => {
      tries++;
      if (document.querySelector(".lp-root") || tries > 40) {
        clearInterval(timer);
        cb();
      }
    }, 250);
  }

  function openRecapSheet(data) {
    const host = getOverlayHost();
    let overlay = document.getElementById("lp-boost-recap-overlay");
    if (!overlay) {
      overlay = document.createElement("div");
      overlay.className = "lp-sheet-overlay";
      overlay.id = "lp-boost-recap-overlay";
      host.appendChild(overlay);
      overlay.addEventListener("click", (e) => { if (e.target === overlay) closeSheet(overlay); });
    }

    const recap = data.recap;
    const confirmed = data.status === "confirmed";
    const scopeText = scopeTextFor(recap.scope, recap.productName);
    const extra = [];
    if (data.chargedNgn) extra.push({ key: "Charged (NGN)", value: formatPrice(data.chargedNgn, "NGN") });
    const rows = rowsFromServerRecap(recap, scopeText, formatDuration(recap.hours), formatPrice(recap.amount, recap.currency), extra);

    overlay.innerHTML = `
      <div class="lp-sheet">
        <div class="lp-sheet-top-bar">
          <div class="lp-sheet-handle"></div>
          <button class="lp-sheet-close" id="lp-boost-recap-close" type="button" aria-label="Close">
            <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg>
          </button>
        </div>
        <h3 class="lpb-title">${confirmed ? "Boost confirmed" : "Payment received"}</h3>
        <div class="lpb-status ${confirmed ? "ok" : "wait"}">${confirmed ? ICON.check + "<span>Your boost is active</span>" : "<span>We're confirming your payment — this can take a minute</span>"}</div>
        ${recapRowsHtml(rows)}
        ${notesHtml(recap.notes)}
        <button type="button" class="lp-btn lp-btn-primary lpb-btn-full" id="lp-boost-recap-done" style="margin-top:6px;"><span>Done</span></button>
      </div>
    `;
    document.getElementById("lp-boost-recap-close").addEventListener("click", () => closeSheet(overlay));
    document.getElementById("lp-boost-recap-done").addEventListener("click", () => closeSheet(overlay));
    openSheet(overlay);
  }

  async function handleReturnFromCheckout() {
    const { result, ref } = RETURN_PARAMS;
    if (!result) return;

    waitForHost(async () => {
      if (result === "success") {
        showToast("Payment confirmed — your boost is being activated!", "success");
        if (ref) {
          try {
            const res = await fetch(`${WORKER_BASE}/api/boost/recap?ref=${encodeURIComponent(ref)}`, { credentials: "include" });
            if (res.ok) {
              const data = await res.json();
              if (data && data.success && data.recap) openRecapSheet(data);
            }
          } catch (e) { /* recap is a bonus — the toast above already told them */ }
        }
      } else {
        showToast("Payment couldn't be confirmed. If you were charged, message us on WhatsApp.", "error");
      }
    });
  }

  handleReturnFromCheckout();

  window.LiyogBoostUI = {
    openBoostSheet: openBoostSheetWithContext
  };
})();
