      var unl = Number(max) < 0, pct = unl ? 8 : Math.min(100, Math.round(usedN / Math.max(1, Number(max)) * 100));
      return '<div class="bz-pl-m"><div class="bz-pl-m-t"><span>' + label + "</span><strong>" + usedN + " / " + lim(max) + '</strong></div><div class="bz-st-bar"><span style="width:' + pct + '%"></span></div></div>';
    }
    view.innerHTML = '<section class="bz-st-card bz-pl-hero"><span class="bz-st-pill gold">Your plan</span><h2>' + esc(cap(pl.planId || "free")) + '</h2><p class="bz-st-note">Upgrade any time. Payment is one-time and your limits go up straight away.</p><button type="button" class="bz-st-btn bz-st-btn-gold" data-act="upgrade">' + ic("plan", 18) + "<span>Upgrade my plan</span></button></section>" +
      '<section class="bz-st-card"><h3 class="bz-st-h3">What you can do</h3>' + meter("Items (products + services)", used, L.max_products == null ? 5 : L.max_products) + meter("Collections", st.collections || 0, L.max_collections == null ? 2 : L.max_collections) +
      '<div class="bz-pl-facts"><div><span>Photos per item</span><strong>' + (L.max_images_per_product == null ? 1 : lim(L.max_images_per_product)) + "</strong></div><div><span>Items per collection</span><strong>" + (L.max_items_per_collection == null ? 25 : lim(L.max_items_per_collection)) + "</strong></div><div><span>Video on items</span><strong>" + (L.video_allowed ? "Included" : "Not included") + "</strong></div></div></section>";
  };

  /* ------------------------------ actions (one delegated listener) ------------------------------ */
  root.addEventListener("click", function (e) {
    var goEl = e.target.closest("[data-go]");
    if (goEl && !e.metaKey && !e.ctrlKey && !e.shiftKey && e.button === 0) { e.preventDefault(); go(goEl.dataset.go); return; }
    var a = e.target.closest("[data-act]"); if (!a) { var bo = e.target.closest("[data-boost]"); if (bo) boostScope(bo.dataset.boost); return; }
    var act = a.dataset.act, p = S.p;
    if (act === "reload") return location.reload();
    if (act === "retry" || act === "refresh") return refreshPage();
    if (act === "add") return openForm(null, function () { if (S.page === "items") mountPage("items"); });
    if (act === "upgrade") { var t = window.LiyogTiersUI; return t ? t.openUpgradeSheet(p) : toast("Upgrades didn't load. Please refresh.", "error"); }
    if (act === "insights") { var bu = window.LiyogBoostUI; return bu && bu.openInsights ? bu.openInsights(p) : toast("Insights didn't load. Please refresh.", "error"); }
    if (act === "share-page") return shareLink(location.origin + "/b/" + p.slug, p.business_name + " | BiziPlex", p.tagline || "Discover " + p.business_name + " on BiziPlex");
    if (act === "copy-ref") { var inp = $("#rf-link"); return copyText(inp ? inp.value : "").then(function () { toast("Link copied!"); }, function () { window.prompt("Copy this link", inp ? inp.value : ""); }); }
    if (act === "share-ref") { var l = $("#rf-link"); return shareLink(l ? l.value : "", "Join me on BiziPlex", "Create your free brand profile on BiziPlex."); }
    if (act === "referrals-list") { var ow = OW(); return ow && ow.openReferralsSheet ? ow.openReferralsSheet(p) : toast("The list didn't load. Please refresh.", "error"); }
  });
  function boostScope(scope) {
    var bu = window.LiyogBoostUI; if (!bu) return toast("Boosting didn't load. Please refresh.", "error");
    bu.openBoostSheet(S.p, { scope: scope });
  }
  root.addEventListener("change", function (e) {
    var sw = e.target.closest("[data-switch]");
    if (sw) { if (S.dirty && !window.confirm("You have changes that aren't saved. Switch anyway?")) { renderSide(); return; } S.dirty = false; location.assign("/studio/" + encodeURIComponent(sw.value)); }
  });
  // pictures that fail to load swap to a calm placeholder instead of a broken icon
  root.addEventListener("error", function (e) {
    var im = e.target; if (!im || im.tagName !== "IMG" || !im.classList.contains("bz-img")) return;
    var ph = document.createElement("span"); ph.className = "bz-it-ph"; ph.innerHTML = ic("items", 26); if (im.parentNode) im.replaceWith(ph);
  }, true);

  /* ------------------------------ network + safety ------------------------------ */
  function net(on) { var el = $("#st-net"); if (el) el.hidden = on; }
  window.addEventListener("offline", function () { net(false); });
  window.addEventListener("online", function () { net(true); toast("You're back online."); if (S.errored) refreshPage(); });
  window.addEventListener("popstate", function () { S.dirty = false; if (S.framed) render(true); });
  window.addEventListener("beforeunload", function (e) { if (S.dirty) { e.preventDefault(); e.returnValue = ""; } });
  window.addEventListener("pageshow", function (e) { if (e.persisted && S.framed) refreshQuietly(); });

  // what the profile editor calls after a successful save (instead of reloading the whole page)
  window.LiyogStudioHooks = {
    afterProfileSave: function (data) {
      S.dirty = false; bust("g:");
      try { sessionStorage.removeItem(ssKey(S.slug)); } catch (e) {}
      if (data && data.newSlug) { location.assign("/studio/" + encodeURIComponent(data.newSlug) + "/profile"); return; }
      refreshQuietly(); toast("Saved! Your changes are live.");
    }
  };

  if (!navigator.onLine) setTimeout(function () { net(false); }, 0);
  boot();
})();
