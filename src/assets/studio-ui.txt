/* =====================================================================
   BIZIPLEX — studio-ui.js  (owner tools inside "Manage My Products & Services")
     LiyogStudio.openProductForm(profile, product|null, onDone)   add / edit a product or service
     LiyogStudio.openCollections(profile, onChange)               create / organise collections
   Reuses what you already have: the same image moderation + upload endpoint, the same WebP compression
   (profile.js compressImageToWebp), the same text moderation on save. The server enforces plan limits,
   ownership and safety; this file only makes the screens friendly. If the form can't load an optional
   part it still opens, and tells you plainly what is missing.
   ===================================================================== */
(function () {
  "use strict";
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var $$ = function (s, r) { return Array.prototype.slice.call((r || document).querySelectorAll(s)); };
  function esc(s) { return String(s == null ? "" : s).replace(/[&<>"']/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]; }); }
  function ui() { return window.LiyogProductUI || {}; }
  function toast(m, t) { if (ui().showToast) ui().showToast(m, t || "success"); }
  function host() { return document.querySelector(".lp-root") || document.body; }

  function api(method, url, body) {
    return fetch(url, { method: method, credentials: "include", headers: body ? { "content-type": "application/json" } : {}, body: body ? JSON.stringify(body) : undefined })
      .then(function (r) {
        return r.text().then(function (t) {
          var d = {}; try { d = t ? JSON.parse(t) : {}; } catch (e) {}
          if (!r.ok) { var er = new Error(d.error || ("Something went wrong (code " + r.status + "). Please try again.")); er.status = r.status; throw er; }
          return d;
        });
      });
  }

    host().appendChild(ov); if (ui().lockBodyScroll) ui().lockBodyScroll();
    var closed = false, onClose = null;
    function close() { if (closed) return; closed = true; ov.remove(); if (ui().unlockBodyScroll) ui().unlockBodyScroll(); if (onClose) onClose(); }
    $(".lp-sheet-close", ov).addEventListener("click", close); ov.addEventListener("click", function (e) { if (e.target === ov) close(); });
    return { ov: ov, body: $(".bz-st-body", ov), close: close, onClose: function (f) { onClose = f; } };
  }

  /* ---------- pictures: crop to an allowed shape (services), compress with YOUR compressor, measure ---------- */
  var RATIOS = [{ k: "1:1", r: 1 }, { k: "4:5", r: 0.8 }, { k: "16:9", r: 16 / 9 }];
  function loadImg(file) {
    return new Promise(function (res, rej) {
      var url = URL.createObjectURL(file), im = new Image();
      im.onload = function () { res({ im: im, url: url }); }; im.onerror = function () { URL.revokeObjectURL(url); rej(new Error("That doesn't look like an image.")); }; im.src = url;
    });
  }
  function cropTo(file, ratio) {
    return loadImg(file).then(function (o) {
      var w = o.im.naturalWidth, h = o.im.naturalHeight, tw = w, th = Math.round(w / ratio);
      if (th > h) { th = h; tw = Math.round(h * ratio); }
      var c = document.createElement("canvas"); c.width = tw; c.height = th;
      c.getContext("2d").drawImage(o.im, (w - tw) / 2, (h - th) / 2, tw, th, 0, 0, tw, th); URL.revokeObjectURL(o.url);
      return new Promise(function (res, rej) { c.toBlob(function (b) { b ? res(new File([b], "crop.jpg", { type: "image/jpeg" })) : rej(new Error("Couldn't prepare that photo.")); }, "image/jpeg", 0.95); });
    });
  }
  function chooseCrop(file, nearest) {
    return loadImg(file).then(function (o) {
      return new Promise(function (resolve) {
        var pick = nearest, dlg = document.createElement("div"); dlg.className = "bz-st-crop";
        dlg.innerHTML = '<div class="bz-st-crop-card" role="dialog" aria-modal="true"><strong>Choose a shape for this service</strong><p class="bz-st-note">Services look cleanest as 1:1, 4:5 or 16:9. We trim equally from the edges. Flyer with text near the edge? Pick the shape that matches your design.</p><div class="bz-st-crop-box" id="bz-cb"><img alt="" src="' + o.url + '"></div><div class="bz-st-crop-ratios">' +
          RATIOS.map(function (x) { return '<button type="button" class="bz-chip' + (x.k === pick.k ? " is-active" : "") + '" data-k="' + x.k + '">' + x.k + "</button>"; }).join("") +
          '</div><div style="display:flex;gap:8px;margin-top:14px"><button type="button" class="bz-pill bz-pill-quiet" id="bz-c-no" style="flex:1;justify-content:center">Cancel</button><button type="button" class="bz-pill" id="bz-c-ok" style="flex:1;justify-content:center;background:var(--liyog-green);color:#fff;border-color:var(--liyog-green)">Use this shape</button></div></div>';
        document.body.appendChild(dlg);
        var box = $("#bz-cb", dlg); function setBox() { box.style.aspectRatio = String(pick.r); } setBox();
        $$(".bz-chip", dlg).forEach(function (b) { b.addEventListener("click", function () { pick = RATIOS.filter(function (x) { return x.k === b.dataset.k; })[0]; $$(".bz-chip", dlg).forEach(function (x) { x.classList.toggle("is-active", x === b); }); setBox(); }); });
        function done(v) { URL.revokeObjectURL(o.url); dlg.remove(); resolve(v); }
        $("#bz-c-no", dlg).addEventListener("click", function () { done(null); });
        $("#bz-c-ok", dlg).addEventListener("click", function () { done(pick); });
      });
    });
  }
  function fallbackCompress(file, max) {
    return loadImg(file).then(function (o) {
      var w = o.im.naturalWidth, h = o.im.naturalHeight, r = Math.min(1, max / Math.max(w, h)); w = Math.round(w * r); h = Math.round(h * r);
      var c = document.createElement("canvas"); c.width = w; c.height = h; c.getContext("2d").drawImage(o.im, 0, 0, w, h); URL.revokeObjectURL(o.url);
      return new Promise(function (res, rej) { c.toBlob(function (b) { b ? res(b) : rej(new Error("Compression failed")); }, "image/webp", 0.82); });
    });
  }
  function measure(blob) {
    return loadImg(blob).then(function (o) {
      var w = o.im.naturalWidth, h = o.im.naturalHeight, color = null;
      try { var t = document.createElement("canvas"); t.width = 1; t.height = 1; var x = t.getContext("2d"); x.drawImage(o.im, 0, 0, 1, 1); var d = x.getImageData(0, 0, 1, 1).data; color = "#" + [d[0], d[1], d[2]].map(function (v) { return ("0" + v.toString(16)).slice(-2); }).join(""); } catch (e) {}
      URL.revokeObjectURL(o.url); return { w: w, h: h, color: color };
    });
  }
  function slugName(s) { return (s || "item").toLowerCase().replace(/[^a-z0-9\s-]/g, "").replace(/\s+/g, "-").slice(0, 40); }
  function uploadBlob(p, blob) {
    var name = ui().buildSeoFilename ? ui().buildSeoFilename(p.business_name, "product") : slugName(p.business_name) + "-product";
    return fetch("/api/upload-product-image?name=" + encodeURIComponent(name), { method: "POST", credentials: "include", headers: { "content-type": "image/webp" }, body: blob })
      .then(function (r) { return r.json().catch(function () { return {}; }).then(function (d) { if (!r.ok) throw new Error(d.error || "Upload failed. Please try again."); return d.url; }); });
  }
  /** pick -> (services: shape) -> compress -> measure -> upload */
  function prepareAndUpload(p, file, kind) {
    var step = Promise.resolve(file);
    if (kind === "service") {
      step = loadImg(file).then(function (o) {
        var ar = o.im.naturalWidth / o.im.naturalHeight; URL.revokeObjectURL(o.url);
        var best = RATIOS.slice().sort(function (a, b) { return Math.abs(Math.log(ar / a.r)) - Math.abs(Math.log(ar / b.r)); })[0];
        if (Math.abs(ar / best.r - 1) <= 0.04) return file;                       // already a clean shape
        return chooseCrop(file, best).then(function (pick) { if (!pick) throw new Error("cancelled"); return cropTo(file, pick.r); });
      });
    }
    return step.then(function (f) {
      var max = kind === "service" ? 1100 : 900;
      return (ui().compressImageToWebp ? ui().compressImageToWebp(f, max) : fallbackCompress(f, max));
    }).then(function (blob) { return measure(blob).then(function (m) { return uploadBlob(p, blob).then(function (url) { return { url: url, w: m.w, h: m.h, color: m.color }; }); }); });
  }

  /* ---------- product / service form ---------- */
  var CURRENCIES = ["NGN", "USD", "GBP", "EUR", "GHS", "KES", "ZAR", "UGX", "TZS", "RWF", "XOF", "XAF", "EGP", "CAD", "AUD", "INR", "AED", "SAR", "BRL", "CNY", "JPY"];
  var ZERO = { JPY: 1, KRW: 1, VND: 1, UGX: 1, RWF: 1, XAF: 1, XOF: 1 };
  function ytId(s) { var m = String(s || "").trim().match(/(?:youtu\.be\/|[?&]v=|\/shorts\/|\/embed\/|\/live\/)([A-Za-z0-9_-]{11})/) || String(s || "").trim().match(/^([A-Za-z0-9_-]{11})$/); return m ? m[1] : null; }
  function settled(pr, fallback) { return pr.then(function (v) { return { ok: true, v: v }; }, function (e) { return { ok: false, e: e, v: fallback }; }); }

  function openProductForm(p, product, onDone) {
    var editing = !!(product && product.id);
    Promise.all([
      settled(api("GET", "/api/profiles/" + p.id + "/plan-limits"), null),
      editing ? settled(api("GET", "/api/products/" + product.id + "/media"), null) : Promise.resolve({ ok: true, v: { images: [], video: null, collections: [], fields: {} } }),
      settled(api("GET", "/api/profiles/" + p.id + "/collections?mine=1"), { collections: [] })
    ]).then(function (r) {
      var problems = [];
      var lim = r[0].ok ? r[0].v : { limits: { max_images_per_product: 1, video_allowed: 0, max_collections: 0 }, defaultCurrency: "NGN" };
      if (!r[0].ok) problems.push("plan limits");
      var media = r[1].ok ? r[1].v : { images: [], video: null, collections: [], fields: {} };
      if (!r[1].ok) problems.push("your saved photos");
      var cols = (r[2].v && r[2].v.collections) || []; if (!r[2].ok) problems.push("collections");
      var maxImg = lim.limits.max_images_per_product < 0 ? 12 : Math.max(1, lim.limits.max_images_per_product);
      var videoOk = !!lim.limits.video_allowed, f = media.fields || {};
      var s = {
        id: editing ? product.id : null, kind: (f.kind || product && product.kind) === "service" ? "service" : "product",
        name: editing && !product.is_draft ? product.name || "" : "", desc: editing ? product.description || "" : "",
        cur: f.currency || lim.defaultCurrency || "NGN", ptype: f.price_type || "fixed",
        price: f.price_minor != null ? String(f.price_minor / Math.pow(10, ZERO[f.currency] ? 0 : 2)) : "",
        images: (media.images || []).slice(), video: media.video ? "https://youtu.be/" + media.video.id : "", hadVideo: !!media.video,
        cols: new Set(media.collections || []), origCols: new Set(media.collections || [])
      };
      var sh = sheet(editing ? "Edit listing" : "Add a product or service");
      sh.onClose(function () { if (onDone) onDone(); });   // a photo may already have created a draft: refresh the list either way

      function grab() { var g = function (id) { var e = $("#" + id, sh.body); return e ? e.value : null; };
        if ($("#st-name", sh.body)) { s.name = g("st-name"); s.desc = g("st-desc"); s.cur = g("st-cur"); s.price = g("st-price"); s.ptype = g("st-ptype"); var v = g("st-video"); if (v !== null) s.video = v; } }
      function status(m, bad) { var e = $("#st-status", sh.body); if (e) { e.textContent = m; e.className = "lp-edit-form-status " + (bad ? "lp-edit-error" : "lp-edit-success"); } }

      function render() {
        var slots = "";
        for (var i = 0; i < maxImg; i++) {
          var im = s.images[i];
          slots += '<div class="bz-st-slot" data-i="' + i + '" role="button" tabindex="0" aria-label="' + (im ? "Photo " + (i + 1) : "Add a photo") + '">' + (im ? '<img src="' + esc(im.url) + '" alt="">' + (i === 0 ? "<em>Cover</em>" : "") + (i > 0 ? '<u data-cover="' + esc(im.id) + '" title="Make cover">★</u>' : "") + '<b data-del="' + esc(im.id) + '" title="Remove">×</b>' : "+ Photo") + "</div>";
        }
        var vid = ytId(s.video);
        sh.body.innerHTML =
          (problems.length ? '<div class="bz-st-tip" id="st-problem">Some options couldn\'t load (' + esc(problems.join(", ")) + '). You can still continue. <button type="button" class="bz-linkbtn" id="st-diag">Check what\'s missing</button></div>' : "") +
          '<div class="bz-st-seg" id="st-kind"><button type="button" data-k="product" class="' + (s.kind === "product" ? "is-active" : "") + '">Product<small>Things you sell</small></button><button type="button" data-k="service" class="' + (s.kind === "service" ? "is-active" : "") + '">Service<small>Flyers &amp; bookings</small></button></div>' +
          '<div class="lp-edit-field"><label class="lp-field-label">Name</label><input id="st-name" class="lp-field-input" maxlength="80" value="' + esc(s.name) + '" placeholder="' + (s.kind === "service" ? "e.g. Logo design" : "e.g. Ankara fabric bundle") + '"></div>' +
          '<div class="lp-edit-field"><label class="lp-field-label">Price</label><div class="bz-st-row"><select id="st-cur" class="lp-field-input">' + CURRENCIES.map(function (c) { return "<option" + (c === s.cur ? " selected" : "") + ">" + c + "</option>"; }).join("") + '</select><input id="st-price" class="lp-field-input" inputmode="decimal" placeholder="15000" value="' + esc(s.price) + '"></div>' +
          '<div class="bz-st-row3"><select id="st-ptype" class="lp-field-input"><option value="fixed"' + (s.ptype === "fixed" ? " selected" : "") + '>Fixed price</option><option value="from"' + (s.ptype === "from" ? " selected" : "") + '>Starting from</option><option value="contact"' + (s.ptype === "contact" ? " selected" : "") + '>Contact for price</option></select></div><p class="bz-st-note">Visitors also see it converted to their own currency.</p></div>' +
          '<div class="lp-edit-field"><label class="lp-field-label">Description (optional)</label><textarea id="st-desc" class="lp-field-input lp-field-textarea" maxlength="500">' + esc(s.desc) + "</textarea></div>" +
          '<div class="lp-edit-field"><label class="lp-field-label">Photos (' + s.images.length + "/" + maxImg + ')</label><div class="bz-st-media" id="st-media">' + slots + '</div><input type="file" accept="image/*" id="st-file" style="display:none">' +
          '<div class="bz-st-tip">' + (s.kind === "service" ? "Services look best as <b>1:1</b>, <b>4:5</b> or <b>16:9</b>. If your picture is a different shape we'll help you pick one. The whole image is shown, never cut." : "Tip: a clean, <b>square (1:1)</b> photo looks best and loads fastest. Any shape works, we fit it neatly.") + " The first photo is the cover" + (maxImg > 1 ? "; tap ★ to change it." : ".") + "</div></div>" +
          '<div class="lp-edit-field"><label class="lp-field-label">YouTube video (optional)</label><input id="st-video" class="lp-field-input" ' + (videoOk ? "" : "disabled ") + 'placeholder="' + (videoOk ? "Paste a YouTube link" : "Video is available on paid plans") + '" value="' + esc(s.video) + '">' +
          (vid ? '<div class="bz-st-vidprev"><img src="https://i.ytimg.com/vi/' + vid + '/default.jpg" alt=""><span>Video found. Visitors tap to play.</span></div>' : "") + "</div>" +
          '<div class="lp-edit-field"><label class="lp-field-label">Collections</label>' +
          (cols.length ? '<div class="bz-st-chips" id="st-cols">' + cols.map(function (c) { return '<button type="button" class="bz-chip' + (s.cols.has(c.id) ? " is-active" : "") + '" data-c="' + esc(c.id) + '">' + esc(c.title) + "</button>"; }).join("") + "</div>" : '<p class="bz-st-note" style="margin:0 0 4px">No collections yet. Create one below to group related items.</p>') +
          '<div class="bz-st-newcoll"><input id="st-newcoll" class="lp-field-input" maxlength="60" placeholder="New collection name"><button type="button" class="bz-pill" id="st-addcoll">＋ Create</button></div></div>' +
          '<div class="lp-edit-form-status" id="st-status"></div><button type="button" class="lp-btn lp-btn-primary lp-submit-btn" id="st-save"><span>' + (editing ? "Save changes" : "Add listing") + "</span></button>";
        wire();
      }

      function wire() {
        var b = sh.body;
        $$("#st-kind button", b).forEach(function (x) { x.addEventListener("click", function () { grab(); s.kind = x.dataset.k; render(); }); });
        $$("#st-cols .bz-chip", b).forEach(function (x) { x.addEventListener("click", function () { var id = x.dataset.c; if (s.cols.has(id)) s.cols.delete(id); else s.cols.add(id); x.classList.toggle("is-active"); }); });
        var dg = $("#st-diag", b); if (dg) dg.addEventListener("click", function () {
          api("GET", "/api/shop/health").then(function (h) { status(h.ok ? "Everything looks fine. Please close and reopen this form." : "Database upgrade incomplete. Missing: " + h.missing.join(", "), !h.ok); })
            .catch(function () { status("The shop service isn't answering. Please make sure the new files were uploaded and the worker was redeployed.", true); });
        });
        var file = $("#st-file", b), pending = -1;
        $$("#st-media .bz-st-slot", b).forEach(function (slot) {
          function act(e) {
            var del = e.target.closest && e.target.closest("[data-del]"), cov = e.target.closest && e.target.closest("[data-cover]");
            if (del) { slot.classList.add("is-busy"); api("DELETE", "/api/products/" + s.id + "/images/" + del.dataset.del).then(function () { s.images = s.images.filter(function (x) { return x.id !== del.dataset.del; }); grab(); render(); }).catch(function (er) { toast(er.message, "error"); grab(); render(); }); return; }
            if (cov) { var ids = s.images.map(function (x) { return x.id; }).filter(function (x) { return x !== cov.dataset.cover; }); ids.unshift(cov.dataset.cover); api("PUT", "/api/products/" + s.id + "/images-order", { image_ids: ids }).then(function () { s.images.sort(function (a, c) { return ids.indexOf(a.id) - ids.indexOf(c.id); }); grab(); render(); }).catch(function (er) { toast(er.message, "error"); }); return; }
            if (s.images[Number(slot.dataset.i)]) return; pending = Number(slot.dataset.i); file.click();
          }
          slot.addEventListener("click", act); slot.addEventListener("keydown", function (e) { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); act(e); } });
        });
        file.onchange = function () {
          var fl = file.files[0]; file.value = ""; if (!fl) return;
          grab(); var slot = $('#st-media .bz-st-slot[data-i="' + pending + '"]', b); if (slot) slot.classList.add("is-busy");
          prepareAndUpload(p, fl, s.kind).then(function (up) {
            if (!s.id) return api("POST", "/api/products", { profile_id: p.id, is_draft: true, kind: s.kind, image_url: up.url, img_w: up.w, img_h: up.h, img_color: up.color }).then(function (d) { s.id = d.productId; s.images.push({ id: d.imageId, url: up.url }); });
            return api("POST", "/api/products/" + s.id + "/images", { url: up.url, w: up.w, h: up.h, color: up.color }).then(function (d) { s.images.push({ id: d.id, url: up.url }); });
          }).then(function () { render(); })
            .catch(function (er) { if (er.message !== "cancelled") { toast(er.message, "error"); status(er.message, true); } if (slot) slot.classList.remove("is-busy"); });
        };
        $("#st-addcoll", b).addEventListener("click", function () {
          var inp = $("#st-newcoll", b), t = inp.value.trim(); if (!t) return;
          api("POST", "/api/profiles/" + p.id + "/collections", { title: t, reach: "brand" }).then(function (d) { cols.push({ id: d.id, title: t }); s.cols.add(d.id); grab(); render(); })
            .catch(function (er) { status(er.message, true); });
        });
        $("#st-save", b).addEventListener("click", save);
      }

      function save() {
        grab(); var name = (s.name || "").trim(); if (!name) return status("Please give it a name.", true);
        var btn = $("#st-save", sh.body); btn.disabled = true; status("Saving…");
        var fields = { name: name, description: (s.desc || "").trim(), kind: s.kind, currency: s.cur, price_type: s.ptype, price: s.ptype === "contact" ? "" : String(s.price || "").trim() };
        (s.id ? api("PATCH", "/api/products/" + s.id, fields) : api("POST", "/api/products", Object.assign({ profile_id: p.id }, fields)).then(function (d) { s.id = d.productId; }))
          .then(function () { var v = (s.video || "").trim(); if (videoOk && (v || s.hadVideo)) return api("PUT", "/api/products/" + s.id + "/video", { video_url: v }); })
          .then(function () {
            var jobs = [];
            s.cols.forEach(function (id) { if (!s.origCols.has(id)) jobs.push(api("PUT", "/api/collections/" + id + "/items", { add: [s.id] })); });
            s.origCols.forEach(function (id) { if (!s.cols.has(id)) jobs.push(api("PUT", "/api/collections/" + id + "/items", { remove: [s.id] })); });
            return Promise.all(jobs);
          })
          .then(function () { toast("Saved!"); sh.close(); })
          .catch(function (er) { btn.disabled = false; status(er.message, true); });
      }
      render();
    });
  }

  /* ---------- collections manager (Manage My Products > Collections) ---------- */

" disabled" : "") + ' aria-label="Move down">↓</button><button class="bz-st-mini" data-a="items">Items</button><button class="bz-st-mini" data-a="edit">Edit</button></div>';
        }).join("") : '<p class="bz-st-note">No collections yet.</p>') + '</div><button type="button" class="lp-btn lp-btn-primary lp-submit-btn" id="co-new" style="margin-top:14px"><span>New collection</span></button>';
      $("#co-new", sh.body).addEventListener("click", function () { edit(null); });
      $$(".bz-st-item", sh.body).forEach(function (row) {
        row.addEventListener("click", function (e) {
          var b = e.target.closest("[data-a]"); if (!b || b.disabled) return; var id = row.dataset.id, a = b.dataset.a, c = cols.filter(function (x) { return x.id === id; })[0];
          if (a === "edit") edit(c); else if (a === "items") items(c);
          else { var ids = cols.map(function (x) { return x.id; }), i = ids.indexOf(id), j = a === "up" ? i - 1 : i + 1; ids.splice(j, 0, ids.splice(i, 1)[0]); api("PUT", "/api/profiles/" + p.id + "/collections-order", { collection_ids: ids }).then(load).then(changed).catch(function (er) { toast(er.message, "error"); }); }
        });
      });
    }
    function edit(c) {
      sh.body.innerHTML = '<button type="button" class="bz-linkbtn" id="co-back">‹ Back</button>' +
        '<div class="lp-edit-field" style="margin-top:10px"><label class="lp-field-label">Title</label><input id="co-title" class="lp-field-input" maxlength="60" value="' + esc(c ? c.title : "") + '" placeholder="e.g. New arrivals"></div>' +
        '<div class="lp-edit-field"><label class="lp-field-label">Description (optional)</label><textarea id="co-desc" class="lp-field-input lp-field-textarea" maxlength="200">' + esc(c ? c.description || "" : "") + "</textarea></div>" +
        '<div class="lp-edit-field"><label class="lp-field-label">Who can discover it</label><select id="co-reach" class="lp-field-input"><option value="brand"' + (c && c.reach === "public" ? "" : " selected") + '>My pages only</option><option value="public"' + (c && c.reach === "public" ? " selected" : "") + ">Public (can appear in Explore later)</option></select></div>" +
        '<label class="lp-review-recommend-toggle"><input type="checkbox" id="co-prof"' + (!c || c.show_on_profile ? " checked" : "") + '> Show as a row on my profile</label>' +
        '<label class="lp-review-recommend-toggle"><input type="checkbox" id="co-live"' + (!c || c.status === "live" ? " checked" : "") + '> Visible to visitors</label>' +
        '<div class="lp-edit-form-status" id="co-status"></div><button type="button" class="lp-btn lp-btn-primary lp-submit-btn" id="co-save"><span>Save</span></button>' +
        (c ? '<button type="button" class="bz-pill bz-pill-quiet" id="co-del" style="margin-top:12px">Delete this collection</button>' : "");
      $("#co-back", sh.body).addEventListener("click", draw);
      $("#co-save", sh.body).addEventListener("click", function () {
        var body = { title: $("#co-title", sh.body).value, description: $("#co-desc", sh.body).value, reach: $("#co-reach", sh.body).value, show_on_profile: $("#co-prof", sh.body).checked ? 1 : 0, status: $("#co-live", sh.body).checked ? "live" : "draft" };
        $("#co-save", sh.body).disabled = true;
        (c ? api("PATCH", "/api/collections/" + c.id, body) : api("POST", "/api/profiles/" + p.id + "/collections", body))
          .then(function (d) { changed(); return load().then(function () { if (!c && d.id) items(cols.filter(function (x) { return x.id === d.id; })[0]); }); })
          .catch(function (er) { $("#co-save", sh.body).disabled = false; var e = $("#co-status", sh.body); e.textContent = er.message; e.className = "lp-edit-form-status lp-edit-error"; });
      });
      var del = $("#co-del", sh.body); if (del) del.addEventListener("click", function () { if (!confirm("Delete this collection? Your products stay.")) return; api("DELETE", "/api/collections/" + c.id).then(load).then(changed).catch(function (er) { toast(er.message, "error"); }); });
    }
    function items(c) {
      if (!c) return draw();
      sh.body.innerHTML = '<p class="bz-st-note">Loading your products…</p>';
      Promise.all([api("GET", "/api/profiles/" + p.id + "/products"), api("GET", "/api/collections/" + c.id + "/members")]).then(function (r) {
        var all = (r[0].products || []).filter(function (x) { return !x.is_draft; }), orig = new Set(r[1].ids), now = new Set(r[1].ids), q = "";
        function list() {
          var shown = all.filter(function (x) { return !q || x.name.toLowerCase().indexOf(q) >= 0; });
          $("#it-list", sh.body).innerHTML = shown.map(function (x) { return '<label class="bz-st-item"><input type="checkbox" data-id="' + esc(x.id) + '"' + (now.has(x.id) ? " checked" : "") + ">" + (x.image_url ? '<img src="' + esc(x.image_url) + '" alt="">' : '<span class="bz-st-ph"></span>') + '<div class="bz-st-t"><strong>' + esc(x.name) + "</strong></div></label>"; }).join("") || '<p class="bz-st-note">No products found.</p>';
          $$("#it-list input", sh.body).forEach(function (cb) { cb.addEventListener("change", function () { cb.checked ? now.add(cb.dataset.id) : now.delete(cb.dataset.id); }); });
        }
        sh.body.innerHTML = '<button type="button" class="bz-linkbtn" id="it-back">‹ Back</button><h4 style="margin:10px 0">' + esc(c.title) + '</h4><input id="it-q" class="lp-field-input" placeholder="Search your products" style="margin-bottom:10px"><div class="bz-st-list" id="it-list"></div><div class="lp-edit-form-status" id="it-status"></div><button type="button" class="lp-btn lp-btn-primary lp-submit-btn" id="it-save" style="margin-top:12px"><span>Save items</span></button>';
        list(); $("#it-back", sh.body).addEventListener("click", draw); $("#it-q", sh.body).addEventListener("input", function (e) { q = e.target.value.toLowerCase(); list(); });
        $("#it-save", sh.body).addEventListener("click", function () {
          var add = [], remove = []; now.forEach(function (id) { if (!orig.has(id)) add.push(id); }); orig.forEach(function (id) { if (!now.has(id)) remove.push(id); });
          if (!add.length && !remove.length) return draw();
          $("#it-save", sh.body).disabled = true;
          api("PUT", "/api/collections/" + c.id + "/items", { add: add, remove: remove }).then(load).then(changed).catch(function (er) { $("#it-save", sh.body).disabled = false; var e = $("#it-status", sh.body); e.textContent = er.message; e.className = "lp-edit-form-status lp-edit-error"; });
        });
      }).catch(function (er) { toast(er.message || "Couldn't load your products", "error"); draw(); });
    }
    load().catch(function (er) { toast(er.message, "error"); sh.close(); });
  }

  window.LiyogStudio = { openProductForm: openProductForm, openCollections: openCollections };
