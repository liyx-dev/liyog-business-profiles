/* =====================================================================
   BIZIPLEX — shop-ui.js  (additions for the product page + profile collection rows)
   Works WITH your existing product-pages-client.js / profile.js (window.LiyogProductUI) — it never
   replaces them. Adds: photo gallery + video player, price in the visitor's currency, search-this-brand
   (cached, paginated), the smart bottom contact bar, and collection rows on the brand profile.
   Everything is optional: if one part can't run, the rest of the page is unaffected.
   ===================================================================== */
(function () {
  "use strict";
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var $$ = function (s, r) { return Array.prototype.slice.call((r || document).querySelectorAll(s)); };
  function esc(s) { return String(s == null ? "" : s).replace(/[&<>"']/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]; }); }
  function getJson(url) { return fetch(url, { credentials: "same-origin" }).then(function (r) { if (!r.ok) throw new Error("HTTP " + r.status); return r.json(); }); }
  var ui = function () { return window.LiyogProductUI || {}; };

  /* ---------------- price in the visitor's own currency (display only) ---------------- */
  var fxPromise = null;
  function loadFx() {
    if (fxPromise) return fxPromise;
    try { var c = JSON.parse(sessionStorage.getItem("bz_fx") || "null"); if (c && Date.now() - c.t < 1800000) return (fxPromise = Promise.resolve(c.d)); } catch (e) {}
    fxPromise = getJson("/api/fx").then(function (d) { try { sessionStorage.setItem("bz_fx", JSON.stringify({ t: Date.now(), d: d })); } catch (e) {} return d; }).catch(function () { return null; });
    return fxPromise;
  }
  var ZERO = { JPY: 1, KRW: 1, VND: 1, UGX: 1, RWF: 1, XAF: 1, XOF: 1, CLP: 1, PYG: 1 };
  function exp(c) { return ZERO[c] ? 0 : (c === "KWD" || c === "BHD" || c === "OMR" || c === "JOD" || c === "TND") ? 3 : 2; }
  function money(minor, cur) {
    var major = minor / Math.pow(10, exp(cur));
    try { return new Intl.NumberFormat(undefined, { style: "currency", currency: cur, currencyDisplay: "narrowSymbol", maximumFractionDigits: major >= 100 ? 0 : exp(cur) }).format(major); } catch (e) { return cur + " " + Math.round(major); }
  }
  function applyFx(root) {
    var els = $$("[data-fx-minor]:not([data-fx-done])", root);
    if (!els.length) return;
    loadFx().then(function (fx) {
      if (!fx || !fx.rates) return;
      var viewer = fx.currency;
      els.forEach(function (el) {
        el.setAttribute("data-fx-done", "1");
        var cur = el.getAttribute("data-fx-cur"), minor = Number(el.getAttribute("data-fx-minor"));
        if (!cur || cur === viewer || !minor) return;
        var rf = cur === "USD" ? 1 : fx.rates[cur], rt = viewer === "USD" ? 1 : fx.rates[viewer];
        if (!rf || !rt) return;
        var out = Math.round(minor / Math.pow(10, exp(cur)) / rf * rt * Math.pow(10, exp(viewer)));
        var orig = el.textContent;
        el.innerHTML = (el.hasAttribute("data-fx-from") ? "From " : "") + "≈ " + esc(money(out, viewer)) + "<small>" + esc(orig) + "</small>";
        el.title = "Approximate. The seller prices this in " + cur + ".";
      });
    });
  }

  /* ---------------- YouTube: tap to play, privacy mode, graceful failure ---------------- */
  var ytPromise = null, players = [];
  function loadYT() {
    if (window.YT && window.YT.Player) return Promise.resolve();
    if (ytPromise) return ytPromise;
    ytPromise = new Promise(function (res, rej) {
      var prev = window.onYouTubeIframeAPIReady; window.onYouTubeIframeAPIReady = function () { if (prev) prev(); res(); };
      var s = document.createElement("script"); s.src = "https://www.youtube.com/iframe_api"; s.async = true; s.onerror = rej; document.head.appendChild(s);
      setTimeout(function () { rej(new Error("timeout")); }, 10000);
    });
    return ytPromise;
  }
  function videoFail(box, id, code, productId) {
    box.innerHTML = '<div class="bz-video-fail"><p>This video can\'t be played here.</p><a href="https://www.youtube.com/watch?v=' + encodeURIComponent(id) + '" target="_blank" rel="noopener noreferrer">Watch on YouTube</a></div>';
    if (productId) fetch("/api/products/" + productId + "/video-report", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ code: code }) }).catch(function () {});
  }
  function initVideos(scope, productId) {
    $$(".bz-video", scope).forEach(function (box) {
      if (box.__bz) return; box.__bz = true;
      var id = box.getAttribute("data-yt"); if (!/^[A-Za-z0-9_-]{11}$/.test(id)) return;
      var btn = $(".bz-video-play", box); if (!btn) return;
      btn.addEventListener("click", function () {
        var holder = document.createElement("div"); holder.style.cssText = "position:absolute;inset:0"; box.innerHTML = ""; box.appendChild(holder);
        loadYT().then(function () {
          var pl = new window.YT.Player(holder, { videoId: id, host: "https://www.youtube-nocookie.com", width: "100%", height: "100%",
            playerVars: { autoplay: 1, rel: 0, playsinline: 1, modestbranding: 1, origin: location.origin },
            events: { onError: function (e) { videoFail(box, id, e.data, productId); } } });
          players.push(pl);
        }).catch(function () { videoFail(box, id, -1, null); });
      });
    });
  }
  function pauseVideos() { players.forEach(function (p) { try { p.pauseVideo(); } catch (e) {} }); }

  /* ---------------- gallery ---------------- */
  function initGallery(ctx) {
    var track = $("#bz-gal-track"); if (!track) return;
    var slides = $$(".bz-gal-slide", track), dots = $$(".bz-gal-dot"), count = $("#bz-gal-count"), n = slides.length;
    function cur() { return Math.round(track.scrollLeft / (track.clientWidth || 1)); }
    function go(i) { i = Math.max(0, Math.min(n - 1, i)); track.scrollTo({ left: i * track.clientWidth, behavior: "smooth" }); }
    var last = 0;
    function mark() {
      var i = cur(); if (count) count.textContent = (i + 1) + " / " + n;
      dots.forEach(function (d, k) { d.classList.toggle("is-active", k === i); });
      if (i !== last) { pauseVideos(); last = i; }
    }
    var tick = false; track.addEventListener("scroll", function () { if (!tick) { tick = true; requestAnimationFrame(function () { tick = false; mark(); }); } }, { passive: true });
    dots.forEach(function (d) { d.addEventListener("click", function () { go(Number(d.getAttribute("data-i"))); }); });
    var p = $(".bz-gal-prev"), nx = $(".bz-gal-next"); if (p) p.addEventListener("click", function () { go(cur() - 1); }); if (nx) nx.addEventListener("click", function () { go(cur() + 1); });
    var chip = $("#bz-vid-chip"); if (chip) chip.addEventListener("click", function () { go(n - 1); });
    track.addEventListener("keydown", function (e) { if (e.key === "ArrowLeft") go(cur() - 1); if (e.key === "ArrowRight") go(cur() + 1); });
    // tap a photo -> the same full-screen viewer the single-photo page already uses
    var lb = $("#lp-product-hero-lightbox"), lbImg = $("#bz-lb-img"), lbClose = $("#lp-product-hero-lightbox-close");
    if (lb && lbImg) {
      track.addEventListener("click", function (e) {
        var im = e.target.closest(".bz-gal-slide:not(.bz-gal-video) img"); if (!im) return;
        lbImg.src = im.currentSrc || im.src; lb.classList.add("lp-open");
      });
      if (lbClose) lbClose.addEventListener("click", function (e) { e.stopPropagation(); lb.classList.remove("lp-open"); });
      lb.addEventListener("click", function (e) { if (e.target === lb) lb.classList.remove("lp-open"); });
      document.addEventListener("keydown", function (e) { if (e.key === "Escape") lb.classList.remove("lp-open"); });
    }
    initVideos(track, ctx.productId);
    mark();
  }

  /* ---------------- smart bottom contact bar ---------------- */
  function initCtaBar(ctx, profile) {
    var bar = $("#bz-ctabar"); if (!bar) return;
    if (profile && ui().wireProductCardActions) ui().wireProductCardActions(bar, profile);
    var ask = $("#bz-ask-btn"); if (ask) ask.addEventListener("click", function () { if (ui().openInquirySheet && profile) ui().openInquirySheet(profile); });
    var inline = $("#lp-detail-contact");
    if (!inline || !("IntersectionObserver" in window)) { bar.classList.add("is-on"); return; }
    var visible = false;
    new IntersectionObserver(function (es) { visible = es[0].isIntersecting; bar.classList.toggle("is-on", !visible); }, { threshold: 0.4 }).observe(inline);
    bar.classList.add("is-on");
  }

  /* ---------------- search this brand ---------------- */
  var memCache = {};
  function cachedGet(url) {
    if (memCache[url]) return Promise.resolve(memCache[url]);
    try { var c = JSON.parse(sessionStorage.getItem("bzs:" + url) || "null"); if (c && Date.now() - c.t < 300000) { memCache[url] = c.d; return Promise.resolve(c.d); } } catch (e) {}
    return getJson(url).then(function (d) { memCache[url] = d; try { sessionStorage.setItem("bzs:" + url, JSON.stringify({ t: Date.now(), d: d })); } catch (e) {} return d; });
  }
  function recentKey(slug) { return "bz_recent_" + slug; }
  function getRecent(slug) { try { return JSON.parse(localStorage.getItem(recentKey(slug)) || "[]").slice(0, 6); } catch (e) { return []; } }
  function saveRecent(slug, q) { q = (q || "").trim(); if (q.length < 2) return; try { var l = getRecent(slug).filter(function (x) { return x !== q; }); l.unshift(q); localStorage.setItem(recentKey(slug), JSON.stringify(l.slice(0, 6))); } catch (e) {} }

  function initSearch(ctx) {
    var open1 = $("#bz-search-open"), fs = $("#bz-fsearch"), sheet = null, lastFocus = null, token = 0, timer = null;
    var state = { q: "", page: 1 };
    var th = $(".lp-shop-topheader");
    if (th) document.documentElement.style.setProperty("--bz-th-h", th.offsetHeight + "px");

    function priceOf(it) { return it.price_display || (it.price_type === "contact" ? "Contact for price" : ""); }
    function productRow(it) {
      return '<a class="bz-ss-row" href="/product/' + encodeURIComponent(it.slug || it.id) + "/" + encodeURIComponent(ctx.brandSlug) + '"><span class="bz-ss-thumb">' +
        (it.image_url ? '<img src="' + esc(it.image_url) + '" alt="" loading="lazy">' : "") +
        (it.has_video ? '<span class="lp-vid-badge"><svg width="10" height="10" viewBox="0 0 24 24" fill="currentColor"><path d="M8 5v14l11-7z"/></svg></span>' : "") +
        '</span><span class="bz-ss-t"><strong>' + esc(it.name) + "</strong><span>" + esc(priceOf(it)) + "</span><small>" + (it.kind === "service" ? "Service" : "Product") + '</small></span><span class="bz-ss-go">›</span></a>';
    }
    function collectionRow(c) {
      return '<button type="button" class="bz-ss-row" data-coll="' + esc(c.id) + '" data-title="' + esc(c.title) + '"><span class="bz-ss-thumb">' + (c.cover_url ? '<img src="' + esc(c.cover_url) + '" alt="" loading="lazy">' : "") +
        '</span><span class="bz-ss-t"><strong>' + esc(c.title) + "</strong><small>Collection · " + c.item_count + ' items</small></span><span class="bz-ss-go">›</span></button>';
    }

    function build() {
      sheet = document.createElement("div"); sheet.className = "bz-ss"; sheet.hidden = true;
      sheet.innerHTML = '<div class="bz-ss-card" role="dialog" aria-modal="true" aria-label="Search ' + esc(ctx.brandName) + '"><div class="bz-ss-head"><div class="bz-ss-field">' +
        '<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.3" stroke-linecap="round"><circle cx="11" cy="11" r="7"/><path d="M21 21l-4.3-4.3"/></svg>' +
        '<input id="bz-ss-input" type="search" placeholder="Search ' + esc(ctx.brandName) + '" autocomplete="off" enterkeyhint="search" aria-label="Search this brand"></div>' +
        '<button type="button" class="bz-ss-close" aria-label="Close">✕</button></div><div class="bz-ss-body" id="bz-ss-body"></div></div>';
      document.body.appendChild(sheet);
      $(".bz-ss-close", sheet).addEventListener("click", close);
      sheet.addEventListener("click", function (e) { if (e.target === sheet) close(); });
      sheet.addEventListener("keydown", function (e) { if (e.key === "Escape") close(); });
      var input = $("#bz-ss-input", sheet);
      input.addEventListener("input", function () { clearTimeout(timer); timer = setTimeout(function () { state.q = input.value.trim(); state.page = 1; run(); }, 250); });
      input.addEventListener("keydown", function (e) { if (e.key === "Enter") { saveRecent(ctx.brandSlug, input.value); var first = $("#bz-ss-body .bz-ss-row", sheet); if (first) first.click(); } });
      $("#bz-ss-body", sheet).addEventListener("click", function (e) {
        var c = e.target.closest("[data-coll]"); if (c) { drill(c.getAttribute("data-coll"), c.getAttribute("data-title")); return; }
        var r = e.target.closest("[data-recent]"); if (r) { input.value = r.getAttribute("data-recent"); state.q = input.value; state.page = 1; run(); return; }
        var pg = e.target.closest("[data-page]"); if (pg && !pg.disabled) { state.page = Number(pg.getAttribute("data-page")); run(); $("#bz-ss-body", sheet).scrollTop = 0; }
        if (e.target.closest("a.bz-ss-row")) saveRecent(ctx.brandSlug, state.q);
      });
    }
    function open() {
      if (!sheet) build();
      lastFocus = document.activeElement; sheet.hidden = false; document.body.style.overflow = "hidden";
      var input = $("#bz-ss-input", sheet); setTimeout(function () { input.focus(); }, 30); run();
    }
    function close() { if (!sheet) return; sheet.hidden = true; document.body.style.overflow = ""; pauseVideos(); if (lastFocus && lastFocus.focus) lastFocus.focus(); }

    function run() {
      var body = $("#bz-ss-body", sheet), my = ++token;
      body.innerHTML = '<p class="bz-ss-loading">Searching…</p>';
      cachedGet("/api/profiles/" + ctx.profileId + "/search?q=" + encodeURIComponent(state.q) + "&page=" + state.page).then(function (d) {
        if (my !== token) return;
        var h = "";
        if (!state.q && state.page === 1) {
          var rec = getRecent(ctx.brandSlug);
          if (rec.length) h += '<div class="bz-ss-label">Recent searches</div><div class="bz-ss-recent">' + rec.map(function (r) { return '<button type="button" data-recent="' + esc(r) + '">' + esc(r) + "</button>"; }).join("") + "</div>";
        }
        if (d.collections && d.collections.length) h += '<div class="bz-ss-label">Collections</div>' + d.collections.map(collectionRow).join("");
        if (d.products && d.products.length) h += '<div class="bz-ss-label">' + (state.q ? "Products &amp; services" : "Popular here") + "</div>" + d.products.map(productRow).join("");
        if (!d.products.length && !(d.collections && d.collections.length)) h += '<p class="bz-ss-empty">Nothing found for “' + esc(state.q) + '”. Try another word.</p>';
        if (d.pages > 1) h += '<div class="bz-ss-pager"><button type="button" data-page="' + (d.page - 1) + '"' + (d.page <= 1 ? " disabled" : "") + '>‹ Prev</button><span>Page ' + d.page + " of " + d.pages + '</span><button type="button" data-page="' + (d.page + 1) + '"' + (d.page >= d.pages ? " disabled" : "") + ">Next ›</button></div>";
        if (d.brandVideo) h += '<div class="bz-ss-label">Watch ' + esc(ctx.brandName) + '</div><div class="bz-ss-video"><div class="bz-video" data-yt="' + esc(d.brandVideo) + '"><button type="button" class="bz-video-play" aria-label="Play brand video"><img class="bz-video-poster" src="https://i.ytimg.com/vi/' + esc(d.brandVideo) + '/hqdefault.jpg" alt="" loading="lazy" referrerpolicy="no-referrer"><span class="bz-video-btn"><svg width="26" height="26" viewBox="0 0 24 24" fill="#fff"><path d="M8 5v14l11-7z"/></svg></span></button></div></div>';
        body.innerHTML = h; initVideos(body, null);
      }).catch(function () { if (my === token) body.innerHTML = '<p class="bz-ss-empty">Couldn\'t search right now. Check your connection and try again.</p>'; });
    }

    function drill(id, title) {
      var body = $("#bz-ss-body", sheet), offset = 0;
      function load(first) {
        cachedGet("/api/collections/" + id + "/items?offset=" + offset).then(function (d) {
          var rows = d.items.map(productRow).join("");
          if (first) body.innerHTML = '<button type="button" class="bz-ss-back" id="bz-ss-back">‹ Back to search</button><div class="bz-ss-label">' + esc(title) + '</div><div id="bz-ss-list"></div><div id="bz-ss-more"></div>';
          $("#bz-ss-list", sheet).insertAdjacentHTML("beforeend", rows || (first ? '<p class="bz-ss-empty">This collection is empty right now.</p>' : ""));
          var more = $("#bz-ss-more", sheet); more.innerHTML = d.next != null ? '<div class="bz-ss-pager"><button type="button" id="bz-ss-loadmore">Load more</button></div>' : "";
          var lm = $("#bz-ss-loadmore", sheet); if (lm) lm.addEventListener("click", function () { offset = d.next; load(false); });
          var bk = $("#bz-ss-back", sheet); if (bk && first) bk.addEventListener("click", run);
        }).catch(function () { body.innerHTML = '<p class="bz-ss-empty">Couldn\'t open this collection. Please try again.</p>'; });
      }
      body.innerHTML = '<p class="bz-ss-loading">Opening…</p>'; load(true);
    }

    if (open1) open1.addEventListener("click", open);
    if (fs) {
      fs.addEventListener("click", open);
      var shown = false;
      function toggle() {
        var inView = open1 ? open1.getBoundingClientRect().bottom > (parseInt(getComputedStyle(document.documentElement).getPropertyValue("--bz-header-h"), 10) || 60) + (th ? th.offsetHeight : 56) : false;
        var on = window.scrollY > 260 && !inView;
        if (on === shown) return; shown = on; fs.hidden = false;
        requestAnimationFrame(function () { fs.classList.toggle("is-on", on); });
        if (on && window.innerWidth < 560) setTimeout(function () { fs.classList.add("is-compact"); }, 3800); else fs.classList.remove("is-compact");
      }
      var t2 = false; window.addEventListener("scroll", function () { if (!t2) { t2 = true; requestAnimationFrame(function () { t2 = false; toggle(); }); } }, { passive: true });
      toggle();
    }
  }

  /* ---------------- brand profile: collection rows (normal product cards) ---------------- */
  function mountProfileCollections(p, isOwner) {
    var anchor = document.getElementById("lp-products-section") || document.getElementById("lp-products-locked-section");
    var u = ui(); if (!anchor || !u.compactCardHtml || document.getElementById("bz-profile-collections")) return;
    getJson("/api/profiles/" + p.id + "/collections?surface=profile&preview=10").then(function (d) {
      var cols = (d.collections || []).filter(function (c) { return c.items && c.items.length; });
      if (!cols.length) return;
      var host = document.createElement("div"); host.id = "bz-profile-collections"; host.className = "bz-pc";
      anchor.parentNode.insertBefore(host, anchor);
      cols.forEach(function (c) {
        var row = document.createElement("section"); row.className = "bz-pc-row";
        row.innerHTML = '<div class="bz-pc-head"><h3>' + esc(c.title) + '</h3><a class="bz-linkbtn" href="/products/' + encodeURIComponent(p.slug) + '">See all ›</a></div><div class="bz-pc-scroll"></div>';
        var sc = $(".bz-pc-scroll", row);
        c.items.forEach(function (it) {
          it._engagement = { stats: { view_count: it.view_count, average_rating: it.avg_rating, rating_count: it.rating_count, like_count: it.like_count }, my_like: false, my_rating: null };
        });
        sc.innerHTML = c.items.map(function (it) { return '<div class="bz-pc-item">' + u.compactCardHtml(it, p, false) + "</div>"; }).join("");
        $$(".bz-pc-item", sc).forEach(function (w) { w.replaceWith(w.firstElementChild); });
        sc.dataset.allProducts = JSON.stringify(c.items);
        u.wireProductCardActions(sc, p);
        host.appendChild(row);
      });
    }).catch(function () {});
  }

  window.LiyogShopUI = { applyFx: applyFx, mountProfileCollections: mountProfileCollections };

  function boot() {
    try {
      applyFx(document);
      var ctx = window.__BZ_PRODUCT__;
      if (ctx) {
        var pc = window.__LIYOG_PRODUCT_PAGE_CONTEXT__ || {};
        initGallery(ctx);
        initCtaBar(ctx, pc.profile);
        initSearch(ctx);
      }
    } catch (e) { console.error("shop-ui:", e); }
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", boot); else boot();
})();
