/* =====================================================================
   LIYOG WORLD BRAND PROFILES — sponsored-ui.js
   The SINGLE reusable module for rendering sponsored/boosted content
   anywhere on the platform — brand profile pages, catalogue/product
   pages today; homepage, blog, and community feed later. Every
   surface calls the same functions, gets the same premium look, and
   any future visual change happens in exactly one file.

   Self-contained on purpose (same reasoning as tiers-ui.js and
   boost-ui.js): profile.js wraps its own code in a private IIFE and
   never exposes internals, so this file cannot reach into it. Where
   this module needs the REAL, rich product card (so a sponsored
   product looks and behaves exactly like a normal catalogue product,
   per instruction), it calls window.LiyogProductUI's already-exposed
   functions — that bridge already exists and is safe to depend on.

   Exposes window.LiyogSponsoredUI with one method per surface:
     .mountProfilesStrip(containerEl, { excludeProfileId, title, category })
     .mountCataloguesStrip(containerEl, { excludeProfileId, title, category })
     .mountProductsStrip(containerEl, { excludeProfileId, title, category })
   Each mounts a fully self-sufficient horizontal-scroll section (with
   arrows, "See More", and correct card wiring) into the given empty
   container element — callers only need to create/position an empty
   <div> and hand it to this module; nothing else.

   3b-3: PAGE CONTEXT. A strip on a page that cannot name its category
   (blog post, homepage, community feed, another site) reads a few short
   public signals from that page — labels/tags, headline, description,
   opening lines — and sends only that short text (never cookies, never
   personal data). The server decides, conservatively, what the page is
   about and shows boosts from that category first. Declarative embed:
     <div data-liyog-ads="products" data-title="Sponsored"></div>
   (also "profiles" / "catalogues"; optional data-limit, data-category,
   data-context, data-labels, data-exclude, data-auto-context="false").
   Programmatic: LiyogSponsoredUI.mountAuto(el, { surface, context,
   labels, category, ... }) — a feed passes each post's own text as
   `context`. An explicit category always beats page text.

   3b-2: a boost whose advertiser switched "Contact buttons" ON arrives
   with a small `cta` object ({ wa, tel }) from the server. Sponsored
   PRODUCT cards then show the real card's own WhatsApp/Call buttons;
   profile/catalogue cards get a WApp/Call row. Without `cta` a card is
   exactly as it was before. Numbers are re-validated here before use.

   3b-1: `category` (optional) is the category of the page the strip sits
   on. It is sent to the server as a plain slug so boosts that chose
   that category are shown first. The viewer's COUNTRY is never sent —
   the server works that out itself, so it can't be spoofed. A page may
   also declare its category once with
   <meta name="liyog-category" content="electronics">.
   ===================================================================== */

(function () {
  "use strict";

  const WORKER_BASE = "https://www.liyogworld.com.ng";

  function esc(str) {
    return String(str == null ? "" : str).replace(/[&<>"']/g, (c) => ({
      "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
    }[c]));
  }
  function escAttr(str) { return esc(str).replace(/"/g, "&quot;"); }

  function getOverlayHost() {
    return document.querySelector(".lp-root") || document.getElementById("liyog-profile-root") || document.body;
  }
  // Body-scroll lock is shared with profile.js's own reference-counted
  // implementation (exposed via window.LiyogProductUI) rather than
  // maintained separately here. This was the actual root cause of the
  // intermittent "scroll frozen after closing the sponsored products
  // modal" report: two INDEPENDENT counters (this module's own, and
  // profile.js's pre-existing one) both wrote to the same
  // document.body.style.overflow with no shared source of truth —
  // whichever one's count happened to hit zero would clear the
  // property even if the other system still had something open,
  // or vice versa. Falls back to a local no-op-safe implementation
  // only if profile.js's version is somehow unavailable (e.g. this
  // module loaded before brands.js, which shouldn't happen given the
  // documented script load order, but this keeps the module from
  // hard-erroring if it ever does).
  function lockBodyScroll() {
    const ui = window.LiyogProductUI;
    if (ui && ui.lockBodyScroll) { ui.lockBodyScroll(); return; }
    document.body.style.overflow = "hidden";
  }
  function unlockBodyScroll() {
    const ui = window.LiyogProductUI;
    if (ui && ui.unlockBodyScroll) { ui.unlockBodyScroll(); return; }
    document.body.style.overflow = "";
  }

  // Uses the REAL toast system already built and correctly positioned
  // in profile.js (#lp-toast-container, fixed bottom-center, proper
  // z-index) via window.LiyogProductUI.showToast — NOT a separate
  // implementation. A separate implementation was tried earlier
  // (creating its own host div with an invented class name,
  // .lp-toast-host) but that class was never actually given any
  // positioning CSS anywhere, so the toast rendered in the DOM with
  // zero visibility — exactly the "clicking Report does nothing
  // visible" bug reported here. Falls back to a minimal correctly-
  // positioned host only if profile.js's version is somehow
  // unavailable, so this module never hard-fails even if load order
  // is ever disrupted.
  function showToast(message, type = "success") {
    const ui = window.LiyogProductUI;
    if (ui && ui.showToast) { ui.showToast(message, type); return; }

    let host = document.getElementById("lp-toast-container");
    if (!host) {
      host = document.createElement("div");
      host.id = "lp-toast-container";
      getOverlayHost().appendChild(host);
    }
    const toast = document.createElement("div");
    toast.className = `lp-toast lp-toast-${type}`;
    toast.textContent = message;
    host.appendChild(toast);
    setTimeout(() => toast.remove(), 4000);
  }

  // ---------------------------------------------------------------------
  // 3b-1: page category for relevance. Only a plain slug is ever sent;
  // the server validates it again and ignores anything unexpected.
  // ---------------------------------------------------------------------
  function resolveCategory(opts) {
    let raw = opts && opts.category;
    if (!raw) {
      const meta = document.querySelector('meta[name="liyog-category"]');
      raw = meta && meta.getAttribute("content");
    }
    const slug = String(raw || "").toLowerCase().trim();
    return /^[a-z0-9_-]{1,40}$/.test(slug) ? slug : "";
  }
  function categoryQuery(category) {
    return category ? `&category=${encodeURIComponent(category)}` : "";
  }

  // ---------------------------------------------------------------------
  // 3b-3: page context. Reads PUBLIC page signals only, once per page
  // load, never throws, and sends at most ~600 characters. On list pages
  // (several posts on one page, e.g. the homepage) labels/description/
  // opening lines belong to many posts at once, so only the headline is
  // used — that avoids guessing a topic from one random post.
  // ---------------------------------------------------------------------
  function cleanText(str, max) {
    return String(str == null ? "" : str)
      .replace(/[\u0000-\u001F\u007F|]+/g, " ")
      .replace(/\s+/g, " ")
      .trim()
      .slice(0, max);
  }

  let cachedPageContext = null;
  function readPageContext() {
    if (cachedPageContext !== null) return cachedPageContext;
    cachedPageContext = "";
    try {
      const txt = (el) => (el && el.textContent ? el.textContent : "");
      const meta = (sel) => { const m = document.querySelector(sel); return m ? (m.getAttribute("content") || "") : ""; };

      const postBodies = document.querySelectorAll(".post-body, .entry-content");
      const singlePost = postBodies.length === 1 || (postBodies.length === 0 && document.querySelectorAll("article").length === 1);

      const labels = [];
      if (singlePost) {
        document.querySelectorAll('.post-labels a, a[rel="tag"], .labels a').forEach((a) => {
          const t = cleanText(txt(a), 40);
          if (t && labels.length < 8) labels.push(t);
        });
        document.querySelectorAll('meta[property="article:tag"], meta[property="article:section"]').forEach((m) => {
          const t = cleanText(m.getAttribute("content"), 40);
          if (t && labels.length < 8) labels.push(t);
        });
        const kw = cleanText(meta('meta[name="keywords"]'), 120);
        if (kw) labels.push(kw);
      }

      const h = document.querySelector("h1.post-title, h1.entry-title, .post-title h1, .post-title, article h1, h1");
      const headline = cleanText(txt(h) || meta('meta[property="og:title"]') || document.title, 140);
      const description = singlePost ? cleanText(meta('meta[name="description"]') || meta('meta[property="og:description"]'), 160) : "";
      const lead = singlePost && postBodies.length === 1 ? cleanText(txt(postBodies[0]), 220) : "";

      const labelText = cleanText(labels.join(", "), 160);
      if (labelText || headline || description || lead) {
        cachedPageContext = [labelText, headline, description, lead].join("|");
      }
    } catch (e) { cachedPageContext = ""; }
    return cachedPageContext;
  }

  /** Explicit context (a feed post's own text, labels) beats page reading;
   *  autoContext:false switches page reading off for a slot. */
  function resolveContextText(opts) {
    if (opts && (typeof opts.context === "string" || Array.isArray(opts.labels))) {
      const labels = Array.isArray(opts.labels) ? cleanText(opts.labels.slice(0, 8).join(", "), 160) : "";
      const text = typeof opts.context === "string" ? cleanText(opts.context, 400) : "";
      if (labels || text) return labels + "|" + text;
    }
    if (opts && opts.autoContext === false) return "";
    return readPageContext();
  }
  function contextQuery(text) {
    return text ? `&ctx=${encodeURIComponent(text)}` : "";
  }

  // ---------------------------------------------------------------------
  // 3b-2: contact buttons. The server only attaches `cta` to boosts that
  // opted in AND already sanitised the numbers; they are checked again
  // here so nothing odd can ever reach an href.
  // ---------------------------------------------------------------------
  function validCta(cta) {
    if (!cta || typeof cta !== "object") return null;
    const wa = typeof cta.wa === "string" && /^\d{7,15}$/.test(cta.wa) ? cta.wa : null;
    const tel = typeof cta.tel === "string" && /^\+?\d{6,16}$/.test(cta.tel) ? cta.tel : null;
    return wa || tel ? { wa, tel } : null;
  }

  const CTA_CALL_ICON = '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="#fff" stroke-width="2" aria-hidden="true"><path d="M22 16.92v3a2 2 0 0 1-2.18 2 19.79 19.79 0 0 1-8.63-3.07 19.5 19.5 0 0 1-6-6A19.79 19.79 0 0 1 2.12 4.18 2 2 0 0 1 4.11 2h3a2 2 0 0 1 2 1.72c.127.96.362 1.903.7 2.81a2 2 0 0 1-.45 2.11L8.09 9.91a16 16 0 0 0 6 6l1.27-1.27a2 2 0 0 1 2.11-.45c.907.338 1.85.573 2.81.7A2 2 0 0 1 22 16.92z"/></svg>';

  /** WApp / Call row for profile + catalogue ad cards. Empty string when
   *  the boost did not opt in — the card is then unchanged. */
  function adCtaRowHtml(profile, linkMode) {
    if (!profile.isSponsored) return "";
    const cta = validCta(profile.cta);
    if (!cta) return "";
    const name = profile.business_name || "this brand";
    const text = linkMode === "catalogue"
      ? `Hi ${name}, I saw your catalogue on Liyog World and I'm interested.`
      : `Hi ${name}, I found your brand profile on Liyog World and I'm interested.`;
    return `
      <div class="lp-ad-cta">
        ${cta.wa ? `<a class="lp-ad-cta-btn lp-ad-cta-wa" href="https://wa.me/${cta.wa}?text=${encodeURIComponent(text)}" target="_blank" rel="noopener noreferrer" aria-label="Message ${escAttr(name)} on WhatsApp">${CTA_WA_ICON}<span>WApp</span></a>` : ""}

      </div>`;
  }

  /** The stand-in "brand" the real product card is built from. For a
   *  sponsored product that opted in, it carries the contact numbers so
   *  the card's OWN WhatsApp/Call buttons appear and work exactly like on
   *  any catalogue card. Otherwise it has none -> no buttons (as before). */
  function fakeProfileFor(product) {
    const profile = { slug: product.profile_slug, business_name: product.profile_business_name, logo_url: product.profile_logo_url };
    const cta = product.isSponsored ? validCta(product.cta) : null;
    if (cta) {
      if (cta.wa) profile.whatsapp_number = cta.wa;
      if (cta.tel) profile.phone_number = cta.tel;
    }
    return profile;
  }

  function fallbackGradientFor(name) {
    let hash = 0;
    const str = name || "?";

    return AD_FALLBACK_GRADIENTS[hash % AD_FALLBACK_GRADIENTS.length];
  }

  function wireStripArrows(sectionId) {
    const scroll = document.getElementById(`${sectionId}-scroll`);
    const prevBtn = document.getElementById(`${sectionId}-prev`);
    const nextBtn = document.getElementById(`${sectionId}-next`);
    if (!scroll || !prevBtn || !nextBtn) return;

    function cardStep() {
      const card = scroll.firstElementChild;
      return card ? card.offsetWidth + 14 : 220;
    }
    function updateArrowState() {
      const maxScroll = scroll.scrollWidth - scroll.clientWidth;
      prevBtn.disabled = scroll.scrollLeft <= 4;
      nextBtn.disabled = scroll.scrollLeft >= maxScroll - 4;
    }
    prevBtn.addEventListener("click", () => { scroll.scrollBy({ left: -cardStep() * 2, behavior: "smooth" }); });
    nextBtn.addEventListener("click", () => { scroll.scrollBy({ left: cardStep() * 2, behavior: "smooth" }); });
    scroll.addEventListener("scroll", updateArrowState, { passive: true });
    window.addEventListener("resize", updateArrowState);

    // Deferred via double rAF rather than called synchronously: right
    // after innerHTML is set, the browser hasn't necessarily completed
    // layout yet, so scroll.scrollWidth/clientWidth can read as 0 or
    // stale — that made the "next" arrow incorrectly compute as
    // already-at-the-end and disable itself immediately, with nothing
    // to ever correct it afterward except a scroll/resize event the
    // user might never trigger. Waiting two animation frames
    // guarantees layout has actually happened first.
    requestAnimationFrame(() => requestAnimationFrame(updateArrowState));

    // Extra safety: cover-photo cards can shift layout width once
    // their (lazy-loaded) images actually finish loading, well after
    // the rAF check above already ran. Re-checking on each image load
    // keeps arrow state accurate without needing a full re-render.
    scroll.querySelectorAll("img").forEach((img) => {
      if (img.complete) return;
      img.addEventListener("load", updateArrowState, { once: true });
    });
  }

  async function mountProfilesStrip(containerEl, opts) {
    if (!containerEl) return;
    const { excludeProfileId = null, title = "You Might Also Like", limit = 8 } = opts || {};
    const category = resolveCategory(opts);
    const contextText = category ? "" : resolveContextText(opts);
    try {
      const res = await fetch(`${WORKER_BASE}/api/discover/profiles?exclude=${encodeURIComponent(excludeProfileId || "")}&limit=${limit}${categoryQuery(category)}${contextQuery(contextText)}`);
      const data = await res.json();
      const profiles = data.profiles || [];
      if (!profiles.length) { containerEl.hidden = true; return; }

      const sectionId = nextSectionId();
      const seeMoreId = `${sectionId}-see-more`;
      containerEl.hidden = false;
      containerEl.className = "lp-section lp-sponsored-strip-section";
      containerEl.innerHTML = stripShellHtml(sectionId, title, profiles.map((p) => profileCardHtml(p, { linkMode: "profile" })).join(""), seeMoreId);
      wireStripArrows(sectionId);

      document.getElementById(seeMoreId).addEventListener("click", () => {
        openMoreModal("profiles", { excludeProfileId, title: "More Brands", category, contextText });
      });
    } catch (e) { containerEl.hidden = true; }
  }

  // ---------------------------------------------------------------------
  // PRODUCTS strip — the one surface that reuses the REAL product card
  // (window.LiyogProductUI.compactCardHtml) rather than a lightweight
  // ad tile, per instruction: a sponsored product should look and
  // behave exactly like any other catalogue product — same price,
  // description, engagement bar, like/rate/zoom/report — with ONLY
  // the sponsored badge + info trigger added. WhatsApp/Call buttons
  // appear ONLY when the advertiser opted in (product.cta, see
  // fakeProfileFor); otherwise the card has none, as before.
  // ---------------------------------------------------------------------
  async function mountProductsStrip(containerEl, opts) {
    if (!containerEl) return;
    const { excludeProfileId = null, title = "Sponsored Products", limit = 8 } = opts || {};
    const category = resolveCategory(opts);
    const contextText = category ? "" : resolveContextText(opts);
    const ui = window.LiyogProductUI;
    if (!ui || !ui.compactCardHtml) { containerEl.hidden = true; return; }

    try {
      const res = await fetch(`${WORKER_BASE}/api/discover/products?exclude=${encodeURIComponent(excludeProfileId || "")}&limit=${limit}${categoryQuery(category)}${contextQuery(contextText)}`);
      const data = await res.json();
      const products = data.products || [];
      if (!products.length) { containerEl.hidden = true; return; }

      // Real view/like/rating numbers, and working like/rate buttons,
      // require the same per-product _engagement data a normal
      // catalogue card fetches — without this, sponsored cards showed
      // zeros and taps didn't register real state. Fetched in
      // parallel, same pattern as the real catalogue grid; a failed
      // fetch for any one product is non-fatal (that card just shows
      // zero engagement rather than blocking the whole strip).
      await fetchEngagementFor(products);

      const sectionId = nextSectionId();
      const seeMoreId = `${sectionId}-see-more`;
      containerEl.hidden = false;
      containerEl.className = "lp-section lp-sponsored-strip-section";
      containerEl.innerHTML = stripShellHtml(sectionId, title, products.map((p) => productCardHtml(p)).join(""), seeMoreId);
      wireStripArrows(sectionId);
      wireProductCards(document.getElementById(`${sectionId}-scroll`), products);

      document.getElementById(seeMoreId).addEventListener("click", () => {
        openMoreModal("products", { excludeProfileId, title: "More Products", category, contextText });
      });
    } catch (e) { containerEl.hidden = true; }
  }

  async function fetchEngagementFor(products) {
    const ui = window.LiyogProductUI;
    const ds = ui && ui.getDeviceSignal ? ui.getDeviceSignal() : "";
    await Promise.all(products.map(async (prod) => {
      try {
        const res = await fetch(`${WORKER_BASE}/api/products/${prod.id}/engagement?ds=${encodeURIComponent(ds)}`, { credentials: "include" });
        prod._engagement = await res.json();
      } catch (e) { prod._engagement = null; }
    }));
  }

  function productCardHtml(product) {
    const ui = window.LiyogProductUI;
    const fakeProfile = fakeProfileFor(product);
    const cardHtml = ui.compactCardHtml(product, fakeProfile, false);
    if (!product.isSponsored) return cardHtml;

    const href = ui.buildProductPageUrl(product, fakeProfile);
    const wrapper = document.createElement("div");
    wrapper.innerHTML = cardHtml.trim();
    const cardEl = wrapper.firstElementChild;
    if (cardEl) {
      cardEl.classList.add("lp-sponsored-product-card");
      // The normal card's own report-flag button is removed for
      // sponsored items specifically — the info popup already has a
      // "Report this ad" action, so keeping both was two overlapping
      // ways to report the same thing, and the two icons visually
      // collided in the same top-right corner. One clear affordance
      // (the info icon) is cleaner and avoids the collision entirely.
      const reportBtn = cardEl.querySelector(".lp-card-report-btn");
      if (reportBtn) reportBtn.remove();
      cardEl.insertAdjacentHTML("afterbegin", adBadgeHtml());
      cardEl.insertAdjacentHTML("beforeend", adInfoTriggerHtml(product.profile_business_name || "This brand", href));
    }
    return wrapper.innerHTML;
  }

  function wireProductCards(scopeEl, products) {
    if (!scopeEl) return;
    const ui = window.LiyogProductUI;
    // Scoped specifically to .lp-prodcard (the actual card root) —
    // NOT the more general [data-product-id] selector, which also
    // matches the inner .lp-engbar element (it carries the same
    // attribute for its own lookup purposes). Matching both meant
    // wireProductCardActions ran twice per visual card, attaching
    // duplicate click listeners that fired together and canceled each
    // other's optimistic UI toggle — the exact cause of "like appears
    // to do nothing" reported here.
    scopeEl.querySelectorAll(".lp-prodcard[data-product-id]").forEach((cardEl) => {
      const product = products.find((prd) => prd.id === cardEl.dataset.productId);
      if (!product) return;
      const fakeProfile = fakeProfileFor(product);
      cardEl.dataset.allProducts = JSON.stringify([product]);
      ui.wireProductCardActions(cardEl, fakeProfile);
    });
  }

  // ---------------------------------------------------------------------
  // "See More" — now a proper centered modal sheet (same family as the
  // ad-info modal), with a responsive grid, not the previous bottom
  // sheet whose edge card was clipping/overlapping. Products here still
  // use the real card + full wiring.
  // ---------------------------------------------------------------------
  let moreModalRequestToken = 0;

  async function openMoreModal(kind, opts) {
    const { excludeProfileId, title } = opts;
    const category = resolveCategory(opts);
    const contextText = category ? "" : (typeof opts.contextText === "string" ? opts.contextText : resolveContextText(opts));
    const myToken = ++moreModalRequestToken; // any earlier in-flight call becomes stale the instant this runs

    let overlay = document.getElementById("lp-sponsored-more-overlay");
    if (!overlay) {
      overlay = document.createElement("div");
      overlay.className = "lp-sponsored-modal";
      overlay.id = "lp-sponsored-more-overlay";
      getOverlayHost().appendChild(overlay);
      overlay.addEventListener("click", (e) => { if (e.target === overlay) closeMoreModal(overlay); });
    }

    overlay.innerHTML = `
      <div class="lp-sponsored-modal-card lp-sponsored-more-card" id="lp-sponsored-more-card">
        <div class="lp-sponsored-more-handle"></div>
        <button type="button" class="lp-sponsored-modal-close" id="lp-sponsored-more-close" aria-label="Close">
          <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><line x1="18" y1="6" x2="6" y2="18"/><line x1="6" y1="6" x2="18" y2="18"/></svg>
        </button>
        <h3 class="lp-sponsored-more-title">${esc(title)}</h3>
        <div id="lp-sponsored-more-grid" class="lp-sponsored-more-grid">
          <p class="lp-sponsored-more-loading">Loading…</p>
        </div>
      </div>
    `;
    document.getElementById("lp-sponsored-more-close").addEventListener("click", () => closeMoreModal(overlay));
    wireSwipeToDismiss(document.getElementById("lp-sponsored-more-card"), overlay);
    overlay.classList.add("lp-open");
    lockBodyScroll();

    const endpoint = kind === "profiles" ? "profiles" : kind === "catalogues" ? "catalogues" : "products";
    try {
      const res = await fetch(`${WORKER_BASE}/api/discover/${endpoint}?exclude=${encodeURIComponent(excludeProfileId || "")}&limit=20${categoryQuery(category)}${contextQuery(contextText)}`);
      const data = await res.json();
      const items = kind === "profiles" ? (data.profiles || []) : kind === "catalogues" ? (data.catalogues || []) : (data.products || []);

      if (kind === "products") await fetchEngagementFor(items);

      // If the user closed this modal (or opened a different one)
      // while the fetches above were still in flight, this response
      // is stale — do nothing further. Critically, this check happens
      // BEFORE touching the DOM or calling any lock/unlock function,
      // so a slow network response can never act on a modal state that
      // has already moved on, which was the root cause of the
      // intermittent scroll-lock-stuck reports on poor connections.
      if (myToken !== moreModalRequestToken) return;

      const gridEl = document.getElementById("lp-sponsored-more-grid");
      if (!gridEl) return;
      if (!items.length) {
        gridEl.innerHTML = `<p class="lp-sponsored-more-loading">Nothing to show right now — check back soon.</p>`;
        return;
      }
      if (kind === "products") {
        gridEl.innerHTML = items.map((p) => productCardHtml(p)).join("");
        wireProductCards(gridEl, items);
      } else {
        const linkMode = kind === "catalogues" ? "catalogue" : "profile";
        gridEl.innerHTML = items.map((p) => profileCardHtml(p, { linkMode })).join("");
      }
    } catch (e) {
      if (myToken !== moreModalRequestToken) return;
      const gridEl = document.getElementById("lp-sponsored-more-grid");
      if (gridEl) gridEl.innerHTML = `<p class="lp-sponsored-more-loading">Couldn't load — check your connection and try again.</p>`;
    }
  }
  function closeMoreModal(overlay) {
    overlay.classList.remove("lp-open");
    unlockBodyScroll();
  }

  /**
   * Swipe-down-to-dismiss for the mobile bottom-sheet variant of the
   * See More modal — desktop's centered modal doesn't need this (mouse
   * users just click outside or the close button), but a bottom sheet
   * on mobile is expected to be swipeable, matching every other sheet
   * already used across the site.
   */
  function wireSwipeToDismiss(cardEl, overlay) {
    if (!cardEl) return;
    let startY = 0;
    let currentY = 0;
    let dragging = false;

    const handle = cardEl.querySelector(".lp-sponsored-more-handle");
    const dragTarget = handle || cardEl;

    dragTarget.addEventListener("touchstart", (e) => {
      // Only initiate the drag from the handle itself, or from the
      // very top of the card — never from inside the scrollable grid
      // content, so normal vertical scrolling of the grid is never
      // hijacked into a dismiss gesture.
      if (e.target !== handle && cardEl.scrollTop > 0) return;
      startY = e.touches[0].clientY;
      dragging = true;
      cardEl.style.transition = "none";
    }, { passive: true });

    dragTarget.addEventListener("touchmove", (e) => {
      if (!dragging) return;
      currentY = e.touches[0].clientY - startY;
      if (currentY > 0) cardEl.style.transform = `translateY(${currentY}px)`;
    }, { passive: true });

    dragTarget.addEventListener("touchend", () => {
      if (!dragging) return;
      dragging = false;
      cardEl.style.transition = "";
      if (currentY > 80) {
        closeMoreModal(overlay);
      }
      cardEl.style.transform = "";
      currentY = 0;
    });
  }

  // ---------------------------------------------------------------------
  // 3b-3: embed anywhere. mountAuto() mounts a strip into ANY element —
  // a blog post, the homepage, a feed slot — wrapping it in the site's
  // theme root (.lp-root) when the host isn't already inside one, so the
  // look and dark mode work outside the brand pages too.
  // ---------------------------------------------------------------------
  const SURFACES = { products: mountProductsStrip, profiles: mountProfilesStrip, catalogues: mountCataloguesStrip };

  function ensureEmbedSlot(hostEl) {
    const slot = document.createElement("div");
    if (hostEl.closest(".lp-root")) {
      hostEl.appendChild(slot);
    } else {
      const root = document.createElement("div");
      root.className = "lp-root lp-embed";
      root.appendChild(slot);
      hostEl.appendChild(root);
    }
    return slot;
  }

  function mountAuto(hostEl, options) {
    if (!hostEl) return;
    const opts = options || {};
    const mount = SURFACES[String(opts.surface || "products").toLowerCase()];
    if (!mount) return;
    return mount(ensureEmbedSlot(hostEl), opts);
  }

  function readEmbedOptions(host) {
    const opts = { surface: host.getAttribute("data-liyog-ads") || "products" };
    const title = host.getAttribute("data-title");
    if (title) opts.title = title;
    const limit = Number(host.getAttribute("data-limit"));
    if (limit >= 1 && limit <= 20) opts.limit = Math.floor(limit);
    const category = host.getAttribute("data-category");
    if (category) opts.category = category;
    const context = host.getAttribute("data-context");
    if (context) opts.context = context;
    const labels = host.getAttribute("data-labels");
    if (labels) opts.labels = labels.split(",").map((x) => x.trim()).filter(Boolean);
    const exclude = host.getAttribute("data-exclude");
    if (exclude) opts.excludeProfileId = exclude;
    if (host.getAttribute("data-auto-context") === "false") opts.autoContext = false;
    return opts;
  }

  // Declarative embeds load lazily: nothing is fetched until the slot is
  // near the screen, so ads never slow a blog post's first paint.
  function initDeclarativeEmbeds() {
    document.querySelectorAll("[data-liyog-ads]").forEach((host) => {
      if (host.dataset.liyogMounted) return;
      host.dataset.liyogMounted = "1";
      const start = () => mountAuto(host, readEmbedOptions(host));
      if ("IntersectionObserver" in window) {
        const io = new IntersectionObserver((entries) => {
          if (entries.some((e) => e.isIntersecting)) { io.disconnect(); start(); }
        }, { rootMargin: "300px 0px" });
        io.observe(host);
      } else {
        start();
      }
    });
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", initDeclarativeEmbeds);
  else initDeclarativeEmbeds();

  window.LiyogSponsoredUI = {
    mountProfilesStrip,
    mountCataloguesStrip,
    mountProductsStrip,
    mountAuto,
    scanEmbeds: initDeclarativeEmbeds
  };
})();
